"use client";

import React, { useEffect, useRef, useCallback, useState } from "react";
import { Play, Pause, Volume2, VolumeX, SkipBack, SkipForward } from "lucide-react";

interface Chunk {
  id: number;
  start: number;
  end: number;
  duration: number;
  clipId: string;
}

interface AudioPlayerProps {
  src: string;
  chunk: Chunk;
  shouldAutoplay?: boolean;
  onError?: (error: string) => void;
  onReady?: () => void;
  onChunkComplete?: (clipId: string) => void;
  onPlayStateChange?: (isPlaying: boolean) => void;
  onTimeUpdate?: (currentTime: number) => void;
  onPlayerRef?: (player: HTMLAudioElement) => void;
  restorePlayState?: boolean;
  preloadedAudio?: HTMLAudioElement | null;
}

const AudioPlayer: React.FC<AudioPlayerProps> = ({
  src,
  chunk,
  shouldAutoplay: _shouldAutoplay = false,
  onError,
  onReady,
  onChunkComplete,
  onPlayStateChange,
  onTimeUpdate,
  onPlayerRef,
  restorePlayState = false,
  preloadedAudio = null,
}) => {
  const audioRef = useRef<HTMLAudioElement | null>(null);
  const currentChunkRef = useRef<Chunk | null>(null);
  const isSeekingRef = useRef(false);
  const playerInitializedRef = useRef(false);
  
  // Store latest callbacks in refs to avoid dependency issues
  const onReadyRef = useRef(onReady);
  const onPlayerRefRef = useRef(onPlayerRef);
  const onTimeUpdateRef = useRef(onTimeUpdate);
  const onPlayStateChangeRef = useRef(onPlayStateChange);
  const onChunkCompleteRef = useRef(onChunkComplete);
  const onErrorRef = useRef(onError);
  
  // Update refs when props change
  onReadyRef.current = onReady;
  onPlayerRefRef.current = onPlayerRef;
  onTimeUpdateRef.current = onTimeUpdate;
  onPlayStateChangeRef.current = onPlayStateChange;
  onChunkCompleteRef.current = onChunkComplete;
  onErrorRef.current = onError;
  
  const [isReady, setIsReady] = useState(false);
  const [isPlaying, setIsPlaying] = useState(false);
  const [currentTime, setCurrentTime] = useState(0);
  const [isLoading, setIsLoading] = useState(true);
  const [isMuted, setIsMuted] = useState(false);
  const [volume, setVolume] = useState(1);

  // Load chunk function
  const loadChunk = useCallback(async (audioElement: HTMLAudioElement, chunkData: Chunk, autoplayAfterLoad = false) => {
    if (!audioElement || isSeekingRef.current) return;
    
    console.log(`Loading audio chunk ${chunkData.id}`, { 
      hasPreloaded: !!preloadedAudio,
      preloadedSrc: preloadedAudio?.src 
    });
    setIsLoading(true);
    
    try {
      // Store current state
      let wasPlaying = false;
      try {
        wasPlaying = !audioElement.paused;
        if (wasPlaying) audioElement.pause();
      } catch {
        console.warn('Could not read/set audio player state');
      }

      // Check if we have a preloaded audio with the same source as we're trying to load
      if (preloadedAudio?.src && preloadedAudio.src === src) {
        console.log('Using preloaded audio for instant loading');
        
        // If the preloaded audio has buffered data, we can use it directly
        if (preloadedAudio.buffered.length > 0) {
          console.log('Preloaded audio has buffered data');
          
          // Set source from preloaded audio
          audioElement.src = preloadedAudio.src;
          
          currentChunkRef.current = chunkData;
          setIsLoading(false);
          
          // Restore play state if needed
          if (autoplayAfterLoad) {
            setTimeout(() => {
              void (async () => {
                try {
                  if (audioElement.paused) {
                    await audioElement.play();
                    console.log('Started playback with preloaded audio');
                  }
                } catch (playError) {
                  console.warn('Failed to start preloaded audio:', playError);
                }
              })();
            }, 100);
          }
          return;
        }
      }

      // Fallback to regular loading
      const currentSrc = audioElement.src;
      const baseSrc = src.split('?')[0];
      const needsNewSource = !currentSrc?.includes(baseSrc ?? '');
      
      if (needsNewSource) {
        console.log('Loading new audio source');
        audioElement.src = src;

        // Wait for metadata
        await new Promise<void>((resolve, reject) => {
          const timeout = setTimeout(() => {
            console.warn('Audio metadata loading timeout, continuing anyway...');
            resolve();
          }, 30000);
          
          const cleanup = () => {
            clearTimeout(timeout);
            try {
              audioElement.removeEventListener('loadedmetadata', onMetadata);
              audioElement.removeEventListener('error', onMetadataError);
            } catch { /* ignore */ }
          };
          
          const onMetadata = () => {
            cleanup();
            const duration = audioElement.duration ?? 0;
            console.log(`Audio duration: ${duration}s, chunk start: ${chunkData.start}s, chunk end: ${chunkData.end}s`);
            if (chunkData.start >= duration) {
              console.warn(`Chunk start time (${chunkData.start}s) is beyond audio duration (${duration}s)`);
            }
            resolve();
          };
          
          const onMetadataError = () => {
            cleanup();
            console.warn('Audio metadata load error, continuing anyway...');
            resolve();
          };
          
          try {
            if (audioElement.readyState >= 1) {
              cleanup();
              const duration = audioElement.duration ?? 0;
              console.log(`Audio duration: ${duration}s, chunk start: ${chunkData.start}s, chunk end: ${chunkData.end}s`);
              resolve();
            } else {
              audioElement.addEventListener('loadedmetadata', onMetadata);
              audioElement.addEventListener('error', onMetadataError);
            }
          } catch (error) {
            cleanup();
            reject(error instanceof Error ? error : new Error('Unknown error'));
          }
        });
      }

      // Physical chunks: Start from beginning (no seeking needed)
      audioElement.currentTime = 0;

      currentChunkRef.current = chunkData;
      setIsLoading(false);
      
      // If we should restore play state, do it after loading is complete
      if (autoplayAfterLoad) {
        try {
          console.log('Restoring play state after chunk load');
          setTimeout(() => {
            void (async () => {
              try {
                if (audioElement.paused) {
                  await audioElement.play();
                  console.log('Play state restored successfully');
                }
              } catch (playError) {
                console.warn('Failed to restore play state (delayed):', playError);
              }
            })();
          }, 200);
        } catch (playError) {
          console.warn('Failed to restore play state:', playError);
        }
      }
      
    } catch (error) {
      console.error('Failed to load audio chunk:', error);
      setIsLoading(false);
      isSeekingRef.current = false;
      onErrorRef.current?.('Failed to load audio chunk');
    }
  }, [src, preloadedAudio]);

  // Initialize audio player
  useEffect(() => {
    if (playerInitializedRef.current) return;
    
    console.log('Creating new audio player');
    playerInitializedRef.current = true;
    
    const audioElement = new Audio();
    audioElement.preload = 'metadata';
    audioElement.controls = false; // We'll create custom controls
    audioRef.current = audioElement;

    // Event handlers
    const handleLoadedData = () => {
      setIsReady(true);
      onPlayerRefRef.current?.(audioElement);
      onReadyRef.current?.();
    };

    const handleLoadStart = () => setIsLoading(true);
    const handleCanPlay = () => setIsLoading(false);
    const handleWaiting = () => setIsLoading(true);
    const handlePlaying = () => setIsLoading(false);

    // Track play/pause state changes and notify parent
    const handlePlay = () => {
      setIsPlaying(true);
      onPlayStateChangeRef.current?.(true);
    };
    
    const handlePause = () => {
      setIsPlaying(false);
      onPlayStateChangeRef.current?.(false);
    };

    // Time tracking for physical chunks
    const handleTimeUpdate = () => {
      if (isSeekingRef.current || !currentChunkRef.current) return;
      
      try {
        const audioTime = audioElement.currentTime;
        if (typeof audioTime !== 'number') return;
        
        const chunk = currentChunkRef.current;
        
        // For physical chunks, audio time IS the relative time (starts at 0)
        const relativeTime = audioTime;
        
        // Check if chunk is completed (listened to the end of the audio file)
        if (relativeTime >= chunk.duration - 0.5) {
          audioElement.pause();
          setCurrentTime(chunk.duration);
          
          // Trigger completion callback
          onChunkCompleteRef.current?.(chunk.clipId);
          return;
        }
        
        const clampedTime = Math.max(0, Math.min(relativeTime, chunk.duration));
        setCurrentTime(clampedTime);
        onTimeUpdateRef.current?.(clampedTime);
      } catch {
        // Silent fail
      }
    };

    // Error handling
    const handleError = () => {
      const error = audioElement.error;
      console.error('Audio error:', error);
      setIsLoading(false);
      onErrorRef.current?.(error?.message ?? 'Audio playback error');
    };

    // Volume change handler
    const handleVolumeChange = () => {
      setVolume(audioElement.volume);
      setIsMuted(audioElement.muted);
    };

    // Add event listeners
    audioElement.addEventListener('loadeddata', handleLoadedData);
    audioElement.addEventListener('loadstart', handleLoadStart);
    audioElement.addEventListener('canplay', handleCanPlay);
    audioElement.addEventListener('waiting', handleWaiting);
    audioElement.addEventListener('playing', handlePlaying);
    audioElement.addEventListener('play', handlePlay);
    audioElement.addEventListener('pause', handlePause);
    audioElement.addEventListener('timeupdate', handleTimeUpdate);
    audioElement.addEventListener('error', handleError);
    audioElement.addEventListener('volumechange', handleVolumeChange);

    return () => {
      // Remove event listeners
      audioElement.removeEventListener('loadeddata', handleLoadedData);
      audioElement.removeEventListener('loadstart', handleLoadStart);
      audioElement.removeEventListener('canplay', handleCanPlay);
      audioElement.removeEventListener('waiting', handleWaiting);
      audioElement.removeEventListener('playing', handlePlaying);
      audioElement.removeEventListener('play', handlePlay);
      audioElement.removeEventListener('pause', handlePause);
      audioElement.removeEventListener('timeupdate', handleTimeUpdate);
      audioElement.removeEventListener('error', handleError);
      audioElement.removeEventListener('volumechange', handleVolumeChange);
      
      playerInitializedRef.current = false;
    };
  }, []);

  // Load chunk when it changes
  useEffect(() => {
    if (!audioRef.current || !chunk || !isReady) return;
    
    // Prevent loading the same chunk multiple times
    if (currentChunkRef.current?.id === chunk.id) {
      return;
    }
    
    const timeoutId = setTimeout(() => {
      if (audioRef.current) {
        void loadChunk(audioRef.current, chunk, restorePlayState);
      }
    }, 100);
    
    return () => clearTimeout(timeoutId);
  }, [chunk, isReady, loadChunk, restorePlayState]);

  // Control functions
  const togglePlay = useCallback(() => {
    if (!audioRef.current) return;
    
    if (isPlaying) {
      audioRef.current.pause();
    } else {
      void audioRef.current.play();
    }
  }, [isPlaying]);

  const seekTo = useCallback((time: number) => {
    if (!audioRef.current || !currentChunkRef.current) return;
    
    const chunk = currentChunkRef.current;
    const clampedTime = Math.max(0, Math.min(time, chunk.duration));
    audioRef.current.currentTime = clampedTime;
    setCurrentTime(clampedTime);
  }, []);

  const skip = useCallback((seconds: number) => {
    if (!audioRef.current || !currentChunkRef.current) return;
    
    const newTime = audioRef.current.currentTime + seconds;
    seekTo(newTime);
  }, [seekTo]);

  const toggleMute = useCallback(() => {
    if (!audioRef.current) return;
    audioRef.current.muted = !audioRef.current.muted;
  }, []);

  const changeVolume = useCallback((newVolume: number) => {
    if (!audioRef.current) return;
    audioRef.current.volume = Math.max(0, Math.min(1, newVolume));
  }, []);

  const formatTime = (seconds: number): string => {
    const hours = Math.floor(seconds / 3600);
    const minutes = Math.floor((seconds % 3600) / 60);
    const secs = Math.floor(seconds % 60);

    if (hours > 0) {
      return `${hours}:${minutes.toString().padStart(2, "0")}:${secs
        .toString()
        .padStart(2, "0")}`;
    }
    return `${minutes}:${secs.toString().padStart(2, "0")}`;
  };

  const progressPercentage = chunk ? (currentTime / chunk.duration) * 100 : 0;

  return (
    <div className="relative w-full bg-gradient-to-br from-gray-900 to-gray-800 rounded-lg overflow-hidden shadow-lg">
      {/* Audio Visualization Background */}
      <div className="absolute inset-0 bg-gradient-to-r from-purple-900/20 to-blue-900/20" />
      
      {/* Main Content */}
      <div className="relative p-6 space-y-4">
        {/* Chunk Info Header */}
        {chunk && (
          <div className="text-center">
            <h3 className="text-lg font-semibold text-white mb-1">
              Chunk {chunk.id}
            </h3>
            <p className="text-gray-300 text-sm">
              {formatTime(currentTime)} / {formatTime(chunk.duration)}
            </p>
          </div>
        )}

        {/* Progress Bar */}
        <div className="space-y-2">
          <div 
            className="relative h-2 bg-gray-700 rounded-full cursor-pointer"
            onClick={(e) => {
              if (!chunk) return;
              const rect = e.currentTarget.getBoundingClientRect();
              const x = e.clientX - rect.left;
              const percentage = x / rect.width;
              const newTime = percentage * chunk.duration;
              seekTo(newTime);
            }}
          >
            <div 
              className="absolute top-0 left-0 h-full bg-gradient-to-r from-purple-500 to-blue-500 rounded-full transition-all duration-150"
              style={{ width: `${progressPercentage}%` }}
            />
            <div 
              className="absolute top-1/2 w-3 h-3 bg-white rounded-full shadow-lg transform -translate-y-1/2 transition-all duration-150"
              style={{ left: `calc(${progressPercentage}% - 6px)` }}
            />
          </div>
        </div>

        {/* Controls */}
        <div className="flex items-center justify-center space-x-4">
          {/* Skip Back */}
          <button
            onClick={() => skip(-10)}
            className="p-2 text-gray-300 hover:text-white transition-colors"
            title="Skip back 10 seconds"
          >
            <SkipBack className="h-5 w-5" />
          </button>

          {/* Play/Pause */}
          <button
            onClick={togglePlay}
            disabled={isLoading}
            className="p-3 bg-gradient-to-r from-purple-600 to-blue-600 hover:from-purple-700 hover:to-blue-700 text-white rounded-full transition-all disabled:opacity-50 disabled:cursor-not-allowed"
          >
            {isLoading ? (
              <div className="w-6 h-6 animate-spin rounded-full border-2 border-white border-t-transparent" />
            ) : isPlaying ? (
              <Pause className="h-6 w-6" />
            ) : (
              <Play className="h-6 w-6 ml-0.5" />
            )}
          </button>

          {/* Skip Forward */}
          <button
            onClick={() => skip(10)}
            className="p-2 text-gray-300 hover:text-white transition-colors"
            title="Skip forward 10 seconds"
          >
            <SkipForward className="h-5 w-5" />
          </button>
        </div>

        {/* Volume Control */}
        <div className="flex items-center justify-center space-x-3">
          <button
            onClick={toggleMute}
            className="p-1 text-gray-300 hover:text-white transition-colors"
          >
            {isMuted || volume === 0 ? (
              <VolumeX className="h-4 w-4" />
            ) : (
              <Volume2 className="h-4 w-4" />
            )}
          </button>
          
          <div className="w-20">
            <input
              type="range"
              min="0"
              max="1"
              step="0.1"
              value={isMuted ? 0 : volume}
              onChange={(e) => {
                const newVolume = parseFloat(e.target.value);
                changeVolume(newVolume);
                if (audioRef.current) {
                  audioRef.current.muted = newVolume === 0;
                }
              }}
              className="w-full h-1 bg-gray-700 rounded-lg appearance-none cursor-pointer slider"
            />
          </div>
        </div>

        {/* Loading Overlay */}
        {isLoading && (
          <div className="absolute inset-0 bg-black bg-opacity-50 flex items-center justify-center">
            <div className="text-white text-center">
              <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-white mx-auto mb-2"></div>
              <div className="text-sm">Loading chunk {chunk?.id}...</div>
            </div>
          </div>
        )}
      </div>

      <style jsx>{`
        .slider::-webkit-slider-thumb {
          appearance: none;
          height: 12px;
          width: 12px;
          border-radius: 50%;
          background: #ffffff;
          cursor: pointer;
          box-shadow: 0 0 2px 0 rgba(0, 0, 0, 0.2);
        }

        .slider::-moz-range-thumb {
          height: 12px;
          width: 12px;
          border-radius: 50%;
          background: #ffffff;
          cursor: pointer;
          border: none;
          box-shadow: 0 0 2px 0 rgba(0, 0, 0, 0.2);
        }
      `}</style>
    </div>
  );
};

export default AudioPlayer;
