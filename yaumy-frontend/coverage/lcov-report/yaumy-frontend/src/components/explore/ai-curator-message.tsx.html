"use client";

import { useState } from "react";
import { Bot, User, Copy, Check, ChevronDown, ChevronUp } from "lucide-react";
import { Button } from "~/components/ui/button";
import { Badge } from "~/components/ui/badge";
import { cn } from "~/lib/utils";
import { motion, AnimatePresence } from "framer-motion";
import AiCuratorLearningPath from "./ai-curator-learning-path";
import ReactMarkdown from "react-markdown";
import { toast } from "sonner";
import SelectableResourceCard from "./selectable-resource-card";
import type { LearningResource } from "~/types/resources";
import type { ConversationMetadata } from "~/types/metadata";

interface Message {
  id: string;
  role: "user" | "assistant";
  content: string;
  metadata?: ConversationMetadata | null;
  createdAt: Date;
}

interface AiCuratorMessageProps {
  message: Message;
  conversationId: string;
  selectedResources: Map<string, LearningResource>;
  onResourceToggle: (resource: LearningResource) => void;
}

export default function AiCuratorMessage({ 
  message, 
  conversationId,
  selectedResources,
  onResourceToggle
}: AiCuratorMessageProps) {
  const [copied, setCopied] = useState(false);
  const [isExpanded, setIsExpanded] = useState(true);

  const copyToClipboard = () => {
    navigator.clipboard.writeText(message.content);
    setCopied(true);
    toast.success("Copied to clipboard");
    setTimeout(() => setCopied(false), 2000);
  };

  const isUser = message.role === "user";
  const hasLearningPath = message.metadata?.type === "learning_path" && message.metadata?.learningPath;
  const hasResources = message.metadata?.resources && message.metadata.resources.length > 0;
  const isClarification = message.metadata?.type === "clarification";

  return (
    <motion.div
      initial={{ opacity: 0, y: 10 }}
      animate={{ opacity: 1, y: 0 }}
      className={cn(
        "flex gap-3",
        isUser ? "flex-row-reverse" : "flex-row"
      )}
    >
      {/* Avatar */}
      <div className={cn(
        "w-8 h-8 rounded-lg flex items-center justify-center flex-shrink-0",
        isUser 
          ? "bg-gray-100" 
          : "bg-gradient-to-br from-purple-500 to-blue-600 text-white"
      )}>
        {isUser ? <User className="h-5 w-5" /> : <Bot className="h-5 w-5" />}
      </div>

      {/* Message Content */}
      <div className={cn(
        "flex-1 max-w-3xl",
        isUser && "flex justify-end"
      )}>
        <div className={cn(
          "rounded-lg p-4",
          isUser 
            ? "bg-gray-100" 
            : "bg-white border border-gray-200"
        )}>
          {/* Message text - only show if no resources or if it's a user message */}
          {(!hasResources || isUser) && (
            <div className="prose prose-sm max-w-none">
              <ReactMarkdown>{message.content}</ReactMarkdown>
            </div>
          )}

          {/* Show brief intro for resource messages */}
          {hasResources && !isUser && (
            <div className="text-sm text-gray-700 font-medium mb-2">
              {message.content}
            </div>
          )}

          {/* Selectable Resources */}
          {hasResources && (
            <div className="mt-4">
              <div className="grid gap-3">
                {message.metadata?.resources?.map((resource: LearningResource, index: number) => (
                  <SelectableResourceCard
                    key={resource.id || `${message.id}-resource-${index}`}
                    resource={{
                      ...resource,
                      id: resource.id || `${message.id}-resource-${index}`,
                    }}
                    isSelected={selectedResources.has(resource.id || `${message.id}-resource-${index}`)}
                    onToggleSelect={onResourceToggle}
                    index={index}
                  />
                ))}
              </div>
            </div>
          )}

          {/* Learning Path (Legacy) */}
          {hasLearningPath && !hasResources && (
            <div className="mt-4">
              <Button
                variant="ghost"
                size="sm"
                onClick={() => setIsExpanded(!isExpanded)}
                className="mb-2"
              >
                {isExpanded ? <ChevronUp className="h-4 w-4 mr-1" /> : <ChevronDown className="h-4 w-4 mr-1" />}
                {isExpanded ? "Hide" : "Show"} Learning Path
              </Button>
              
              <AnimatePresence>
                {isExpanded && (
                  <motion.div
                    initial={{ height: 0, opacity: 0 }}
                    animate={{ height: "auto", opacity: 1 }}
                    exit={{ height: 0, opacity: 0 }}
                    transition={{ duration: 0.2 }}
                  >
                    <AiCuratorLearningPath
                      learningPath={message.metadata?.learningPath}
                      conversationId={conversationId}
                      messageId={message.id}
                    />
                  </motion.div>
                )}
              </AnimatePresence>
            </div>
          )}

          {/* Clarification Options */}
          {isClarification && (message.metadata as any)?.clarificationQuestion?.options && (
            <div className="mt-3 space-y-2">
              <p className="text-sm font-medium text-gray-700">Choose an option:</p>
              <div className="flex flex-wrap gap-2">
                {(message.metadata as any)?.clarificationQuestion?.options?.map((option: string, index: number) => (
                  <Badge
                    key={index}
                    variant="outline"
                    className="cursor-pointer hover:bg-gray-100"
                  >
                    {option}
                  </Badge>
                ))}
              </div>
            </div>
          )}

          {/* Actions */}
          {!isUser && (
            <div className="flex items-center gap-2 mt-3 pt-3 border-t">
              <Button
                variant="ghost"
                size="sm"
                onClick={copyToClipboard}
                className="h-8"
              >
                {copied ? (
                  <>
                    <Check className="h-3 w-3 mr-1" />
                    Copied
                  </>
                ) : (
                  <>
                    <Copy className="h-3 w-3 mr-1" />
                    Copy
                  </>
                )}
              </Button>
            </div>
          )}
        </div>
      </div>
    </motion.div>
  );
}
