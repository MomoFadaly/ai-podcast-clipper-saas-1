'use client'

import { useEffect, useRef, useMemo } from 'react'
import { Canvas, useFrame, useThree } from '@react-three/fiber'
import { Float, Html } from '@react-three/drei'
import * as THREE from 'three'
import { useAnimationStore } from '~/store/animationStore'

interface ContentFloaterProps {
  item: {
    id: string
    type: 'pdf' | 'video' | 'audio' | 'book' | 'article'
    title: string
    progress: number
    position: { x: number; y: number; z: number }
    rotation: { x: number; y: number; z: number }
    scale: number
    opacity: number
  }
  phase: 'chaos' | 'organizing' | 'organized'
}

function ContentFloater({ item, phase }: ContentFloaterProps) {
  const meshRef = useRef<THREE.Mesh>(null)
  const textRef = useRef<THREE.Group>(null)
  
  // Content type colors and geometries
  const contentConfig = useMemo(() => {
    const configs = {
      pdf: { color: '#ef4444', geometry: 'box' },
      video: { color: '#3b82f6', geometry: 'sphere' },
      audio: { color: '#10b981', geometry: 'cylinder' },
      book: { color: '#f59e0b', geometry: 'box' },
      article: { color: '#8b5cf6', geometry: 'plane' }
    }
    return configs[item.type]
  }, [item.type])

  // Animation based on phase
  useFrame((state) => {
    if (!meshRef.current || !textRef.current) return

    const time = state.clock.elapsedTime
    
    if (phase === 'chaos') {
      // Chaotic floating movement
      meshRef.current.position.x = item.position.x + Math.sin(time * 0.5 + item.position.x * 0.01) * 20
      meshRef.current.position.y = item.position.y + Math.cos(time * 0.3 + item.position.y * 0.01) * 15
      meshRef.current.position.z = item.position.z + Math.sin(time * 0.7 + item.position.z * 0.01) * 10
      
      // Rotation
      meshRef.current.rotation.x = item.rotation.x + time * 0.2
      meshRef.current.rotation.y = item.rotation.y + time * 0.3
      meshRef.current.rotation.z = item.rotation.z + time * 0.1
      
      // Scale pulsing
      const pulseScale = item.scale + Math.sin(time * 2) * 0.1
      meshRef.current.scale.setScalar(pulseScale)
    } else if (phase === 'organizing') {
      // Smooth transition to organized state
      const targetX = (item.id.charCodeAt(0) % 5 - 2) * 100
      const targetY = Math.floor(parseInt(item.id) / 5) * 80 - 100
      const targetZ = 0
      
      meshRef.current.position.x = THREE.MathUtils.lerp(meshRef.current.position.x, targetX, 0.02)
      meshRef.current.position.y = THREE.MathUtils.lerp(meshRef.current.position.y, targetY, 0.02)
      meshRef.current.position.z = THREE.MathUtils.lerp(meshRef.current.position.z, targetZ, 0.02)
      
      // Stabilize rotation
      meshRef.current.rotation.x = THREE.MathUtils.lerp(meshRef.current.rotation.x, 0, 0.05)
      meshRef.current.rotation.y = THREE.MathUtils.lerp(meshRef.current.rotation.y, 0, 0.05)
      meshRef.current.rotation.z = THREE.MathUtils.lerp(meshRef.current.rotation.z, 0, 0.05)
      
      meshRef.current.scale.setScalar(THREE.MathUtils.lerp(meshRef.current.scale.x, 1, 0.03))
    } else {
      // Organized: gentle hovering
      meshRef.current.position.y += Math.sin(time + item.position.x * 0.01) * 0.5
    }
  })

  const geometry = useMemo(() => {
    switch (contentConfig.geometry) {
      case 'sphere':
        return <sphereGeometry args={[1, 16, 16]} />
      case 'cylinder':
        return <cylinderGeometry args={[0.8, 0.8, 1.5, 12]} />
      case 'plane':
        return <planeGeometry args={[1.5, 2]} />
      default:
        return <boxGeometry args={[1.2, 1.6, 0.3]} />
    }
  }, [contentConfig.geometry])

  return (
    <Float speed={phase === 'chaos' ? 2 : 0.5} rotationIntensity={phase === 'chaos' ? 0.5 : 0.1}>
      <mesh ref={meshRef} position={[item.position.x, item.position.y, item.position.z]}>
        {geometry}
        <meshStandardMaterial 
          color={contentConfig.color}
          transparent
          opacity={item.opacity}
          roughness={0.7}
          metalness={0.1}
        />
      </mesh>
      
      <group ref={textRef} position={[item.position.x, item.position.y - 2, item.position.z]}>
        <Html center distanceFactor={10}>
          <div className="text-white text-xs font-medium bg-black/50 px-2 py-1 rounded backdrop-blur-sm">
            {item.title.slice(0, 20)}...
            <div className="w-full bg-gray-700 rounded-full h-1 mt-1">
              <div 
                className="bg-white h-1 rounded-full transition-all duration-300"
                style={{ width: `${item.progress}%` }}
              />
            </div>
          </div>
        </Html>
      </group>
    </Float>
  )
}

function Scene() {
  const { camera } = useThree()
  const { contentItems, graveyardPhase } = useAnimationStore()

  useEffect(() => {
    // Set up camera position
    camera.position.set(0, 0, 300)
    camera.lookAt(0, 0, 0)
  }, [camera])

  return (
    <>
      <ambientLight intensity={0.6} />
      <pointLight position={[100, 100, 100]} intensity={1} />
      <pointLight position={[-100, -100, -100]} intensity={0.5} color="#8b5cf6" />
      
      <fog attach="fog" args={['#000000', 200, 800]} />
      
      {contentItems.map((item) => (
        <ContentFloater key={item.id} item={item} phase={graveyardPhase} />
      ))}
    </>
  )
}

interface ContentGraveyardProps {
  className?: string
}

export function ContentGraveyard({ className = '' }: ContentGraveyardProps) {
  const containerRef = useRef<HTMLDivElement>(null)
  const { graveyardPhase, prefersReducedMotion } = useAnimationStore()

  // Disable 3D if user prefers reduced motion
  if (prefersReducedMotion) {
    return (
      <div className={`relative w-full h-full bg-gradient-to-br from-black via-gray-900 to-black ${className}`}>
        <div className="absolute inset-0 flex items-center justify-center">
          <div className="text-center">
            <h3 className="text-2xl font-bold text-white mb-4">
              Your Content Library
            </h3>
            <p className="text-gray-300 max-w-md">
              Organize your learning materials into digestible chunks for better focus and completion rates.
            </p>
          </div>
        </div>
      </div>
    )
  }

  return (
    <div 
      ref={containerRef} 
      className={`relative w-full h-full bg-gradient-to-br from-black via-gray-900 to-black ${className}`}
    >
      <Canvas
        camera={{ fov: 75, near: 0.1, far: 1000 }}
        className="w-full h-full"
      >
        <Scene />
      </Canvas>
      
      {/* Overlay UI */}
      <div className="absolute inset-0 pointer-events-none">
        <div className="absolute top-8 left-8 text-white">
          <h2 className="text-3xl font-bold mb-2">
            {graveyardPhase === 'chaos' && "The Content Graveyard"}
            {graveyardPhase === 'organizing' && "Organizing Your Learning"}
            {graveyardPhase === 'organized' && "Ready to Learn"}
          </h2>
          <p className="text-gray-300 max-w-md">
            {graveyardPhase === 'chaos' && "All the content you started but never finished, floating in digital limbo."}
            {graveyardPhase === 'organizing' && "Watch as Yaumy transforms chaos into organized, digestible learning paths."}
            {graveyardPhase === 'organized' && "Your content, perfectly chunked and ready for focused learning sessions."}
          </p>
        </div>
        
        <div className="absolute bottom-8 right-8 text-white text-sm">
          Phase: {graveyardPhase}
        </div>
      </div>
    </div>
  )
}
