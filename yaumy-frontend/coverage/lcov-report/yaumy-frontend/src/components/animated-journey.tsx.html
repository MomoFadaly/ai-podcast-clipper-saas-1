"use client";

import { useState, useEffect } from "react";
import { 
  Upload, FileText, Video, Headphones, FileImage, Zap, 
  CheckCircle, Play, Clock, BarChart3, BookOpen, Sparkles
} from "lucide-react";

const AnimatedJourney = () => {
  const [currentStep, setCurrentStep] = useState(1);
  const [isAnimating, setIsAnimating] = useState(false);

  useEffect(() => {
    const interval = setInterval(() => {
      setIsAnimating(true);
      setTimeout(() => {
        setCurrentStep((prev) => (prev === 3 ? 1 : prev + 1));
        setIsAnimating(false);
      }, 500);
    }, 4000);

    return () => clearInterval(interval);
  }, []);

  const contentTypes = [
    { icon: Video, label: "Video", color: "from-red-500 to-pink-500" },
    { icon: FileText, label: "PDF", color: "from-blue-500 to-cyan-500" },
    { icon: Headphones, label: "Podcast", color: "from-green-500 to-emerald-500" },
    { icon: BookOpen, label: "Text", color: "from-purple-500 to-indigo-500" },
    { icon: FileImage, label: "Document", color: "from-orange-500 to-yellow-500" }
  ];

  const chunks = [
    { id: 1, title: "Introduction to ML", duration: "4:30", progress: 100, color: "bg-green-400" },
    { id: 2, title: "Data Preprocessing", duration: "5:15", progress: 100, color: "bg-green-400" },
    { id: 3, title: "Model Training", duration: "6:20", progress: 75, color: "bg-blue-400" },
    { id: 4, title: "Evaluation Metrics", duration: "4:45", progress: 0, color: "bg-gray-600" },
    { id: 5, title: "Deployment", duration: "5:30", progress: 0, color: "bg-gray-600" }
  ];

  return (
    <div className="relative w-full max-w-6xl mx-auto">
      {/* Step Navigation */}
      <div className="flex justify-center mb-16 relative">
        <div className="flex items-center space-x-8">
          {[1, 2, 3].map((step) => (
            <div key={step} className="flex items-center">
              <div className={`
                w-16 h-16 rounded-full flex items-center justify-center text-white font-bold text-xl
                transition-all duration-500 relative overflow-hidden
                ${currentStep === step 
                  ? step === 1 ? 'bg-gradient-to-br from-purple-500 to-pink-500 scale-110' 
                    : step === 2 ? 'bg-gradient-to-br from-blue-500 to-cyan-500 scale-110'
                    : 'bg-gradient-to-br from-green-500 to-emerald-500 scale-110'
                  : 'bg-gray-700 scale-100'
                }
              `}>
                <span className="relative z-10">{step}</span>
                {currentStep === step && (
                  <div className="absolute inset-0 animate-pulse bg-gradient-to-br from-white/20 to-transparent rounded-full"></div>
                )}
              </div>
              {step < 3 && (
                <div className={`
                  w-16 h-1 mx-4 transition-all duration-500
                  ${currentStep > step ? 'bg-gradient-to-r from-purple-400 to-blue-400' : 'bg-gray-600'}
                `}></div>
              )}
            </div>
          ))}
        </div>
      </div>

      {/* Animation Container */}
      <div className="premium-card p-8 backdrop-blur-xl min-h-[400px] relative overflow-hidden">
        <div className={`transition-all duration-500 ${isAnimating ? 'opacity-0 scale-95' : 'opacity-100 scale-100'}`}>
          
          {/* Step 1: Content Upload */}
          {currentStep === 1 && (
            <div className="flex flex-col lg:flex-row items-center justify-between h-full">
              <div className="lg:w-1/2 mb-8 lg:mb-0">
                <h3 className="text-3xl font-bold text-white mb-4 flex items-center">
                  <Upload className="w-8 h-8 mr-3 text-purple-400" />
                  Add Your Content
                </h3>
                <p className="text-gray-300 text-lg leading-relaxed mb-6">
                  Upload any learning material - videos, PDFs, podcasts, documents, or text. 
                  Our AI instantly recognizes and prepares your content for optimal chunking.
                </p>
                <div className="text-sm text-purple-300">
                  ✨ Supports all major formats • Instant processing • Secure upload
                </div>
              </div>
              <div className="lg:w-1/2 flex justify-center">
                <div className="grid grid-cols-3 gap-4 p-8 border-2 border-dashed border-purple-400/50 rounded-2xl bg-purple-500/10">
                  {contentTypes.map((type, index) => {
                    const Icon = type.icon;
                    return (
                      <div 
                        key={index}
                        className={`
                          w-20 h-20 rounded-xl bg-gradient-to-br ${type.color} 
                          flex items-center justify-center text-white
                          animate-bounce relative
                        `}
                        style={{ animationDelay: `${index * 0.2}s`, animationDuration: '2s' }}
                      >
                        <Icon className="w-8 h-8" />
                        <div className="absolute -bottom-6 left-1/2 transform -translate-x-1/2 text-xs text-gray-400 whitespace-nowrap">
                          {type.label}
                        </div>
                      </div>
                    );
                  })}
                </div>
              </div>
            </div>
          )}

          {/* Step 2: AI Processing */}
          {currentStep === 2 && (
            <div className="flex flex-col items-center justify-center h-full text-center">
              <div className="relative mb-8">
                <div className="w-32 h-32 bg-gradient-to-br from-blue-500 to-cyan-500 rounded-full flex items-center justify-center mx-auto relative overflow-hidden">
                  <Zap className="w-16 h-16 text-white animate-pulse" />
                  <div className="absolute inset-0 bg-gradient-to-br from-white/20 to-transparent rounded-full animate-spin"></div>
                </div>
                <div className="absolute -inset-4 rounded-full border-2 border-blue-400/30 animate-ping"></div>
              </div>
              <h3 className="text-3xl font-bold text-white mb-4">
                AI Magic in Progress
              </h3>
              <p className="text-gray-300 text-lg leading-relaxed max-w-2xl mb-6">
                Our advanced neural networks analyze your content&apos;s structure, identify natural learning boundaries, 
                and create perfectly timed segments optimized for maximum retention.
              </p>
              <div className="flex items-center space-x-6 text-sm text-cyan-300">
                <div className="flex items-center">
                  <div className="w-2 h-2 bg-cyan-400 rounded-full animate-pulse mr-2"></div>
                  Analyzing content structure
                </div>
                <div className="flex items-center">
                  <div className="w-2 h-2 bg-cyan-400 rounded-full animate-pulse mr-2" style={{ animationDelay: '0.5s' }}></div>
                  Creating optimal chunks
                </div>
                <div className="flex items-center">
                  <div className="w-2 h-2 bg-cyan-400 rounded-full animate-pulse mr-2" style={{ animationDelay: '1s' }}></div>
                  Generating timestamps
                </div>
              </div>
            </div>
          )}

          {/* Step 3: Chunked Timeline */}
          {currentStep === 3 && (
            <div className="flex flex-col lg:flex-row items-center justify-between h-full">
              <div className="lg:w-1/2 mb-8 lg:mb-0">
                <h3 className="text-3xl font-bold text-white mb-4 flex items-center">
                  <BarChart3 className="w-8 h-8 mr-3 text-green-400" />
                  Track Your Progress
                </h3>
                <p className="text-gray-300 text-lg leading-relaxed mb-6">
                  Your content is now organized into bite-sized chunks with individual progress tracking. 
                  Complete each segment at your own pace and watch your learning momentum build.
                </p>
                <div className="flex items-center space-x-4 text-sm">
                  <div className="flex items-center text-green-300">
                    <CheckCircle className="w-4 h-4 mr-1" />
                    <span>2 completed</span>
                  </div>
                  <div className="flex items-center text-blue-300">
                    <Play className="w-4 h-4 mr-1" />
                    <span>1 in progress</span>
                  </div>
                  <div className="flex items-center text-gray-400">
                    <Clock className="w-4 h-4 mr-1" />
                    <span>2 remaining</span>
                  </div>
                </div>
              </div>
              <div className="lg:w-1/2">
                <div className="space-y-3 bg-black/20 p-6 rounded-2xl border border-white/10">
                  <div className="text-center mb-4">
                    <h4 className="text-white font-semibold">Machine Learning Course</h4>
                    <div className="text-sm text-gray-400">5 chunks • 26:20 total</div>
                  </div>
                  {chunks.map((chunk, index) => (
                    <div 
                      key={chunk.id} 
                      className={`
                        flex items-center p-3 rounded-lg border transition-all duration-500
                        ${chunk.progress === 100 ? 'bg-green-500/10 border-green-500/30' 
                          : chunk.progress > 0 ? 'bg-blue-500/10 border-blue-500/30'
                          : 'bg-gray-800/50 border-gray-600/30'}
                      `}
                      style={{ animationDelay: `${index * 0.1}s` }}
                    >
                      <div className={`w-3 h-3 rounded-full mr-3 ${chunk.color}`}></div>
                      <div className="flex-1">
                        <div className="flex justify-between items-center">
                          <span className="text-white text-sm font-medium">{chunk.title}</span>
                          <span className="text-gray-400 text-xs">{chunk.duration}</span>
                        </div>
                        {chunk.progress > 0 && (
                          <div className="mt-1 w-full bg-gray-700 rounded-full h-1">
                            <div 
                              className={`h-1 rounded-full transition-all duration-1000 ${
                                chunk.progress === 100 ? 'bg-green-400' : 'bg-blue-400'
                              }`}
                              style={{ width: `${chunk.progress}%` }}
                            ></div>
                          </div>
                        )}
                      </div>
                    </div>
                  ))}
                </div>
              </div>
            </div>
          )}
        </div>

        {/* Sparkle Effects */}
        <div className="absolute top-4 right-4">
          <Sparkles className="w-6 h-6 text-purple-400 animate-pulse" />
        </div>
        <div className="absolute bottom-4 left-4">
          <Sparkles className="w-4 h-4 text-blue-400 animate-pulse" style={{ animationDelay: '1s' }} />
        </div>
      </div>

      {/* Step Labels */}
      <div className="flex justify-between mt-8 text-center">
        <div className={`transition-all duration-500 ${currentStep === 1 ? 'text-purple-400 scale-105' : 'text-gray-500'}`}>
          <div className="font-semibold">Upload Content</div>
          <div className="text-xs mt-1">Any format, instant recognition</div>
        </div>
        <div className={`transition-all duration-500 ${currentStep === 2 ? 'text-blue-400 scale-105' : 'text-gray-500'}`}>
          <div className="font-semibold">AI Processing</div>
          <div className="text-xs mt-1">Intelligent chunking & analysis</div>
        </div>
        <div className={`transition-all duration-500 ${currentStep === 3 ? 'text-green-400 scale-105' : 'text-gray-500'}`}>
          <div className="font-semibold">Track Progress</div>
          <div className="text-xs mt-1">Organized timeline with stats</div>
        </div>
      </div>
    </div>
  );
};

export default AnimatedJourney;
