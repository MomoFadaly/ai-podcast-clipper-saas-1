'use client';

/**
 * Linked Note Block Component
 * Display component for linked note blocks
 */

import React from 'react';
import { type BlockComponentProps } from '../../core/ports/block-registry';
import { type LinkedNoteBlockContent } from '../../core/domain/block';
import { ContentLinkType } from '../../core/domain/types';
import { cn } from '~/lib/utils';
import { usePickWhip } from '~/contexts/pick-whip-context';

export const LinkedNoteBlockComponent: React.FC<BlockComponentProps<LinkedNoteBlockContent>> = ({
  block,
  isEditing,
  isSelected,
  isReadOnly,
  onUpdate,
  onDelete,
  onDuplicate,
  onConvert,
  onNavigate
}) => {
  const { content, properties } = block;
  const { link } = content;
  const { getConnectionsForBlock, removeConnection, hoveredConnectionId, setHoveredConnectionId } = usePickWhip();
  
  // Get pick whip connections for this block
  const connections = getConnectionsForBlock(block.id, block.properties);

  const formatLinkDisplay = () => {
    switch (link.type) {
      case ContentLinkType.Video:
      case ContentLinkType.Audio:
        return link.label || formatTimestamp(link.timestamp || 0);
      case ContentLinkType.PDF:
        return `Page ${link.page || 1}`;
      case ContentLinkType.Text:
        return link.label || 'Text selection';
      default:
        return 'Linked';
    }
  };

  const handleLinkClick = () => {
    if (onNavigate) {
      onNavigate(link);
    }
  };

  return (
    <div 
      className={cn(
        "relative group",
        isSelected && "ring-2 ring-primary/50 ring-offset-2",
        properties.alignment && `text-${properties.alignment}`
      )}
      style={{
        color: properties.color,
        backgroundColor: properties.backgroundColor,
        fontSize: properties.fontSize
      }}
    >
      {/* Link indicator */}
      <button
        onClick={handleLinkClick}
        className={cn(
          "inline-flex items-center gap-1 px-2 py-0.5 rounded-md",
          "bg-primary/10 text-primary hover:bg-primary/20",
          "text-xs font-medium mb-2 transition-colors"
        )}
      >
        <svg className="w-3 h-3" fill="none" stroke="currentColor" viewBox="0 0 24 24">
          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M13.828 10.172a4 4 0 00-5.656 0l-4 4a4 4 0 105.656 5.656l1.102-1.101m-.758-4.899a4 4 0 005.656 0l4-4a4 4 0 00-5.656-5.656l-1.1 1.1" />
        </svg>
        {formatLinkDisplay()}
      </button>

      {/* Note content */}
      <p className={cn(
        "whitespace-pre-wrap",
        !content.text && "text-muted-foreground"
      )}>
        {content.text || 'Type your note...'}
      </p>

      {/* Pick Whip Connections */}
      {connections.length > 0 && (
        <div className="mt-2 flex flex-wrap gap-2">
          {connections.map((connection) => (
            <div
              key={connection.id}
              className={cn(
                "inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full",
                "bg-blue-100 text-blue-800 hover:bg-blue-200 border border-blue-300",
                "text-sm font-medium transition-all cursor-pointer group shadow-sm",
                "hover:shadow-md hover:scale-105",
                hoveredConnectionId === connection.id && "ring-2 ring-blue-400"
              )}
              onMouseEnter={() => setHoveredConnectionId(connection.id)}
              onMouseLeave={() => setHoveredConnectionId(null)}
              onClick={() => {
                if (onNavigate) {
                  if (connection.targetType === 'video') {
                    onNavigate({
                      type: ContentLinkType.Video,
                      timestamp: connection.targetPosition.timestamp
                    });
                  } else if (connection.targetType === 'pdf') {
                    onNavigate({
                      type: ContentLinkType.PDF,
                      page: connection.targetPosition.pageNumber
                    });
                  }
                }
              }}
            >
              <svg className="w-3 h-3" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} 
                  d="M12 2C6.48 2 2 6.48 2 12s4.48 10 10 10c1.1 0 2-.9 2-2s-.9-2-2-2c-3.31 0-6-2.69-6-6s2.69-6 6-6 6 2.69 6 6c0 1.1.9 2 2 2s2-.9 2-2c0-5.52-4.48-10-10-10zm0 4c-2.21 0-4 1.79-4 4s1.79 4 4 4 4-1.79 4-4-1.79-4-4-4zm0 2c1.1 0 2 .9 2 2s-.9 2-2 2-2-.9-2-2 .9-2 2-2z" />
              </svg>
              <span>
                {connection.targetType === 'video' && 
                  `Linked to ${formatTimestamp(connection.targetPosition.timestamp || 0)}`}
                {connection.targetType === 'pdf' && 
                  `Linked to page ${connection.targetPosition.pageNumber || 1}`}
              </span>
              <button
                onClick={(e) => {
                  e.stopPropagation();
                  removeConnection(connection.id);
                }}
                className="opacity-60 group-hover:opacity-100 ml-auto hover:text-red-600 transition-opacity"
                title="Remove connection"
              >
                <svg className="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" />
                </svg>
              </button>
            </div>
          ))}
        </div>
      )}

      {/* Block actions - shown on hover */}
      {!isReadOnly && !isEditing && (
        <div className="absolute -left-8 top-0 opacity-0 group-hover:opacity-100 transition-opacity">
          <div className="flex flex-col gap-1">
            <button
              onClick={onDuplicate}
              className="p-1 hover:bg-accent rounded"
              title="Duplicate block"
            >
              <svg className="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M8 16H6a2 2 0 01-2-2V6a2 2 0 012-2h8a2 2 0 012 2v2m-6 12h8a2 2 0 002-2v-8a2 2 0 00-2-2h-8a2 2 0 00-2 2v8a2 2 0 002 2z" />
              </svg>
            </button>
            <button
              onClick={onDelete}
              className="p-1 hover:bg-destructive/20 rounded"
              title="Delete block"
            >
              <svg className="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16" />
              </svg>
            </button>
          </div>
        </div>
      )}
    </div>
  );
};

// Helper function
function formatTimestamp(seconds: number): string {
  const hours = Math.floor(seconds / 3600);
  const minutes = Math.floor((seconds % 3600) / 60);
  const secs = Math.floor(seconds % 60);
  
  if (hours > 0) {
    return `${hours}:${minutes.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
  }
  return `${minutes}:${secs.toString().padStart(2, '0')}`;
}
