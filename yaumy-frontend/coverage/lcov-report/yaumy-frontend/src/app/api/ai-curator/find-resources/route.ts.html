export async function POST(req: NextRequest) {
  try {
    const session = await auth();
    if (!session?.user?.id) {
      return new Response("Unauthorized", { status: 401 });
    }

    const body = await req.json();
    const { query, parameters } = requestSchema.parse(body);

    // Get resources for the topic
    const resources = getResourcesForTopic(query, parameters);

    // Format as streaming response
    const encoder = new TextEncoder();
    const stream = new ReadableStream({
      async start(controller) {
        const response = `I found ${resources.length} great resources for learning about ${query}:\n\n`;
        
        controller.enqueue(encoder.encode(`data: ${JSON.stringify({ text: response })}\n\n`));

        for (let i = 0; i < resources.length; i++) {
          const r = resources[i];
          const text = `**${i + 1}. ${r.title}**\n- Creator: ${r.creator}\n- Type: ${r.type}\n- Duration: ${r.duration}\n- URL: ${r.url}\n- About: ${r.description}\n\n`;
          
          // Stream each character
          for (const char of text) {
            controller.enqueue(encoder.encode(`data: ${JSON.stringify({ text: char })}\n\n`));
            await new Promise(resolve => setTimeout(resolve, 5));
          }
        }

        // Send resources metadata
        controller.enqueue(encoder.encode(`data: ${JSON.stringify({ 
          done: true,
          resources: resources 
        })}\n\n`));
        
        controller.close();
      },
    });

    return new Response(stream, {
      headers: {
        "Content-Type": "text/event-stream",
        "Cache-Control": "no-cache",
        "Connection": "keep-alive",
      },
    });
  } catch (error) {
    console.error("Error finding resources:", error);
    return new Response("Internal Server Error", { status: 500 });
  }
}
