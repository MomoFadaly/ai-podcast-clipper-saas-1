import { type NextRequest, NextResponse } from "next/server";
import { z } from "zod";
import { createTrackedOpenAI } from "~/lib/ai-providers/openai-tracked";
import { auth } from "~/server/auth";

// Request schema
const requestSchema = z.object({
  query: z.string(),
  options: z.object({
    skillLevel: z.enum(["beginner", "intermediate", "advanced", "not_sure"]),
    timeCommitment: z.union([z.string(), z.number()]),
    dailyTime: z.string(),
    preferredFormat: z.enum(["video", "article", "interactive", "mixed"]),
    language: z.string(),
    freeOnly: z.boolean(),
  }),
  clarificationAnswer: z.string().optional(),
  userId: z.string().optional(), // Make optional since we'll get from session
});

export async function POST(req: NextRequest) {
  // Get session
  const session = await auth();
  if (!session?.user?.id) {
    return NextResponse.json(
      { error: "Unauthorized" },
      { status: 401 }
    );
  }
  
  // Check if OpenAI API key is available
  if (!process.env.OPENAI_API_KEY) {
    console.error("OPENAI_API_KEY is not set");
    return NextResponse.json(
      { error: "OpenAI API key is not configured" },
      { status: 500 }
    );
  }

  try {
    const body = await req.json();
    console.log("AI Curator generate request:", { query: body.query, options: body.options });
    
    const { query, options, clarificationAnswer } = requestSchema.parse(body);

    // Convert time commitment to days
    let timeInDays = 7; // Default value
    if (typeof options.timeCommitment === "number") {
      timeInDays = options.timeCommitment;
    } else if (typeof options.timeCommitment === "string") {
      const commitment = options.timeCommitment;
      const parts = commitment.split("_");
      const amount = parts[0] ? parseInt(parts[0]) : 1;
      const multiplier = commitment.includes("week") ? 7 : commitment.includes("month") ? 30 : 1;
      timeInDays = amount * multiplier;
    }

    // Build the system prompt
    const systemPrompt = `You are an expert learning curator AI. Your job is to create personalized learning paths for users based on their queries and preferences.

IMPORTANT: Be VERY smart and make intelligent assumptions:
- Claude, GPT, Gemini, etc. are AI language models - assume user wants to learn about AI/ML/LLMs
- If someone mentions ANY AI model name, they likely want to learn about:
  * How AI/LLMs work
  * Building with AI APIs
  * Prompt engineering
  * AI application development
- Make smart assumptions based on context - don't ask for clarification unless ABSOLUTELY necessary
- Only clarify for extremely vague single-word queries like just "programming" or "design"

MAKE ASSUMPTIONS, DON'T ASK QUESTIONS:
- "Claude Opus 4" → Create learning path about advanced AI/LLMs, prompt engineering, and building AI applications
- "React Native" → Mobile app development with React Native
- "blockchain" → Blockchain technology fundamentals and smart contracts
- "machine learning" → ML fundamentals, Python libraries, and practical projects

ONLY ask clarification for:
- Single vague words: "programming" → "Which language: Python, JavaScript, or Java?"
- Contradictory info: "advanced beginner React" → "Are you a beginner or do you have some React experience?"
- Truly ambiguous: "design" → "UI/UX design, graphic design, or system design?"

User Preferences:
- Skill Level: ${options.skillLevel}
- Time Commitment: ${timeInDays} days
- Daily Time: ${options.dailyTime} minutes
- Preferred Format: ${options.preferredFormat}
- Language: ${options.language}
- Budget: ${options.freeOnly ? "Free resources only" : "Free and paid resources"}

Your response should be a JSON object with the following structure:
{
  "needsClarification": boolean,
  "clarificationQuestion": {
    "question": "string",
    "options": ["option1", "option2", ...] // optional, for multiple choice
  },
  "title": "string",
  "description": "string",
  "estimatedTime": "string (e.g., '4 weeks')",
  "outcomes": ["outcome1", "outcome2", ...],
  "resources": [
    {
      "title": "string",
      "type": "video" | "article" | "course" | "documentation" | "tutorial" | "book",
      "url": "string",
      "creator": "string",
      "duration": "string (e.g., '45 min')",
      "description": "string",
      "difficulty": "beginner" | "intermediate" | "advanced",
      "isFree": boolean,
      "thumbnail": "string (optional)",
      "aiReason": "string (why this resource was chosen)"
    }
  ]
}

Rules:
1. ALWAYS make smart assumptions - only clarify if truly impossible to guess intent
2. For ANY AI model name (Claude, GPT, Llama, etc) → Generate AI/LLM learning resources
3. Return 5-8 high-quality SPECIFIC resources for the assumed topic
4. Include practical tutorials, official docs, and hands-on projects
5. Order by difficulty progression
6. Keep descriptions concise and relevant
7. Include 3-5 specific, achievable outcomes

CRITICAL: Default to creating a learning path, NOT asking questions:
- "Claude Opus 4" → AI/LLM resources (how LLMs work, using Claude API, prompt engineering)
- "GPT stuff" → Resources about GPT models, OpenAI API, building AI apps
- "xyz framework" → Assume it's a tech framework and provide general programming resources

For AI-related queries specifically include:
- How LLMs work (conceptual understanding)
- API documentation and tutorials
- Prompt engineering guides
- Building AI applications
- Best practices and ethics`;

    // Build the user message
    let userMessage = `Learning Query: "${query}"`;
    if (clarificationAnswer) {
      userMessage += `\n\nClarification Answer: "${clarificationAnswer}"`;
    }

    // Create tracked OpenAI client
    const openai = createTrackedOpenAI(session.user.id, {
      feature: 'ai-curator',
      action: 'generate-learning-path',
      query: query
    });
    
    // Call OpenAI with optimized settings
    const completion = await openai.createChatCompletion({
      model: "gpt-4-turbo-preview", // Better knowledge, worth the extra seconds
      messages: [
        { role: "system", content: systemPrompt },
        { role: "user", content: userMessage },
      ],
      temperature: 0.7,
      max_tokens: 2000, // Limit response size for faster generation
      response_format: { type: "json_object" },
    });

    if (!("choices" in completion)) {
      throw new Error("Unexpected API response format");
    }
    const response = JSON.parse(completion.choices[0]?.message?.content || "{}");

    // If clarification is needed, return early
    if (response.needsClarification) {
      return NextResponse.json({
        needsClarification: true,
        clarificationQuestion: response.clarificationQuestion,
      });
    }

    // Return the full learning path
    return NextResponse.json({
      needsClarification: false,
      title: response.title,
      description: response.description,
      estimatedTime: response.estimatedTime,
      outcomes: response.outcomes,
      resources: response.resources,
    });
  } catch (error) {
    console.error("Error generating learning path:", error);
    
    // Provide more detailed error information
    if (error instanceof z.ZodError) {
      return NextResponse.json(
        { error: "Invalid request data", details: error.errors },
        { status: 400 }
      );
    }
    
    if (error instanceof Error) {
      return NextResponse.json(
        { error: "Failed to generate learning path", details: error.message },
        { status: 500 }
      );
    }
    
    return NextResponse.json(
      { error: "Failed to generate learning path" },
      { status: 500 }
    );
  }
}
