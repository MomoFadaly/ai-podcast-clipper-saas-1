import { redirect } from "next/navigation";
import { auth } from "~/server/auth";
import { AnalyticsDashboard } from "~/components/analytics/analytics-dashboard";
import { getAnalyticsData } from "~/actions/analytics";

export default async function AnalyticsPage() {
  const session = await auth();
  
  if (!session?.user) {
    redirect("/login");
  }

  // Only admin and super admin can access analytics dashboard
  if (session.user.role !== "ADMIN" && session.user.role !== "SUPER_ADMIN") {
    redirect("/dashboard");
  }

  let analyticsData;
  try {
    analyticsData = await getAnalyticsData();
  } catch (error) {
    console.error("Failed to load analytics data:", error);
    // Return with empty/default data structure
    analyticsData = {
      traffic: {
        pageViews: { current: 0, previous: 0, change: 0 },
        sessions: { current: 0, previous: 0, change: 0 },
        visitors: { current: 0, previous: 0, change: 0 },
        bounceRate: 0,
        trend: [],
      },
      sources: {
        utmSources: [],
        referrers: [{ name: "Direct", count: 0 }],
        countries: [],
      },
      devices: {
        devices: [],
        browsers: [{ name: "Unknown", count: 0 }],
        operatingSystems: [],
        mobileDesktop: [{ name: "Desktop", count: 0 }],
      },
      topPages: [],
      realTime: {
        activeUsers: 0,
        recentPageViews: 0,
        recentEvents: 0,
      },
      generatedAt: new Date().toISOString(),
    };
  }

  return (
    <div className="space-y-6 p-6">
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-3xl font-bold tracking-tight">Analytics Dashboard</h1>
          <p className="text-muted-foreground">
            Comprehensive traffic analysis, user behavior, and performance metrics
          </p>
        </div>
        <div className="flex items-center gap-3">
          <a
            href="https://us.posthog.com/project/188150/web"
            target="_blank"
            rel="noopener noreferrer"
            className="inline-flex items-center gap-2 px-4 py-2 text-sm font-medium text-white bg-blue-600 hover:bg-blue-700 rounded-lg transition-colors"
          >
            <svg className="h-4 w-4" viewBox="0 0 24 24" fill="currentColor">
              <path d="M12 2L2 7v10c0 5.55 3.84 9.95 9 11 5.16-1.05 9-5.45 9-11V7l-10-5z"/>
            </svg>
            PostHog Deep Dive
            <svg className="h-3 w-3" fill="none" stroke="currentColor" viewBox="0 0 24 24">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M10 6H6a2 2 0 00-2 2v10a2 2 0 002 2h10a2 2 0 002-2v-4M14 4h6m0 0v6m0-6L10 14" />
            </svg>
          </a>
        </div>
      </div>
      
      <AnalyticsDashboard data={analyticsData} />
    </div>
  );
}
