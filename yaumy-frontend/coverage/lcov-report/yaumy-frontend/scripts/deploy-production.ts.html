#!/usr/bin/env node
/**
 * Production Deployment Script
 * 
 * Automated deployment with safety checks and rollback
 */

import { execSync } from 'child_process';
import { readFileSync, writeFileSync } from 'fs';
import { join } from 'path';
import prompts from 'prompts';
import { log } from '../src/lib/logger';

// Deployment configuration
const config = {
  checks: {
    required: [
      'validate:all',
      'build',
      'test:run',
    ],
    health: {
      endpoint: `${process.env.NEXT_PUBLIC_APP_URL}/api/health`,
      timeout: 30000,
      retries: 5,
    },
  },
  notifications: {
    slack: process.env.SLACK_WEBHOOK_URL,
    email: process.env.DEPLOYMENT_EMAIL,
  },
};

// Color codes for console
const colors = {
  reset: '\x1b[0m',
  bright: '\x1b[1m',
  green: '\x1b[32m',
  red: '\x1b[31m',
  yellow: '\x1b[33m',
  blue: '\x1b[34m',
};

// Deployment checklist
const DEPLOYMENT_CHECKLIST = [
  { id: 'env', label: 'Environment variables updated', required: true },
  { id: 'migrations', label: 'Database migrations ready', required: true },
  { id: 'backup', label: 'Database backup completed', required: true },
  { id: 'monitoring', label: 'Monitoring alerts configured', required: true },
  { id: 'ssl', label: 'SSL certificates valid', required: true },
  { id: 'cdn', label: 'CDN cache cleared', required: false },
  { id: 'feature-flags', label: 'Feature flags reviewed', required: true },
  { id: 'rollback', label: 'Rollback plan ready', required: true },
  { id: 'load-test', label: 'Load testing completed', required: false },
  { id: 'security', label: 'Security scan passed', required: true },
];

// Run command with output
function runCommand(command: string, silent = false): string {
  if (!silent) {
    console.log(`${colors.blue}▶ Running:${colors.reset} ${command}`);
  }
  
  try {
    const output = execSync(command, { encoding: 'utf-8', stdio: silent ? 'pipe' : 'inherit' });
    return output;
  } catch (error: any) {
    console.error(`${colors.red}✗ Command failed:${colors.reset} ${command}`);
    throw error;
  }
}

// Check prerequisites
async function checkPrerequisites() {
  console.log(`\n${colors.bright}🔍 Checking prerequisites...${colors.reset}\n`);
  
  const checks = [
    { name: 'Node.js version', command: 'node --version', expected: 'v20' },
    { name: 'Git status', command: 'git status --porcelain', expected: '' },
    { name: 'Current branch', command: 'git branch --show-current', expected: 'main' },
  ];
  
  for (const check of checks) {
    const result = runCommand(check.command, true).trim();
    
    if (check.expected && !result.includes(check.expected)) {
      console.error(`${colors.red}✗ ${check.name}:${colors.reset} Expected ${check.expected}, got ${result}`);
      return false;
    }
    
    console.log(`${colors.green}✓ ${check.name}:${colors.reset} ${result}`);
  }
  
  return true;
}

// Run validation checks
async function runValidationChecks() {
  console.log(`\n${colors.bright}🧪 Running validation checks...${colors.reset}\n`);
  
  for (const check of config.checks.required) {
    try {
      runCommand(`npm run ${check}`);
      console.log(`${colors.green}✓ ${check} passed${colors.reset}`);
    } catch (error) {
      console.error(`${colors.red}✗ ${check} failed${colors.reset}`);
      return false;
    }
  }
  
  return true;
}

// Interactive checklist
async function runChecklist() {
  console.log(`\n${colors.bright}📋 Deployment Checklist${colors.reset}\n`);
  
  const responses = await prompts(
    DEPLOYMENT_CHECKLIST.map(item => ({
      type: 'confirm',
      name: item.id,
      message: item.label,
      initial: false,
    }))
  );
  
  const failed = DEPLOYMENT_CHECKLIST.filter(
    item => item.required && !responses[item.id]
  );
  
  if (failed.length > 0) {
    console.error(`\n${colors.red}✗ Required checks failed:${colors.reset}`);
    failed.forEach(item => console.error(`  - ${item.label}`));
    return false;
  }
  
  return true;
}

// Get deployment metadata
function getDeploymentMetadata() {
  const gitCommit = runCommand('git rev-parse HEAD', true).trim();
  const gitBranch = runCommand('git branch --show-current', true).trim();
  const timestamp = new Date().toISOString();
  
  return {
    version: process.env.npm_package_version || 'unknown',
    commit: gitCommit,
    branch: gitBranch,
    timestamp,
    deployer: process.env.USER || 'unknown',
    environment: 'production',
  };
}

// Create deployment tag
async function createDeploymentTag(metadata: any) {
  const tagName = `deploy-${metadata.version}-${Date.now()}`;
  const tagMessage = JSON.stringify(metadata, null, 2);
  
  runCommand(`git tag -a ${tagName} -m '${tagMessage}'`);
  runCommand(`git push origin ${tagName}`);
  
  return tagName;
}

// Deploy to Vercel
async function deployToVercel(metadata: any) {
  console.log(`\n${colors.bright}🚀 Deploying to Vercel...${colors.reset}\n`);
  
  // Build deployment command
  const envVars = [
    `NEXT_PUBLIC_APP_VERSION=${metadata.version}`,
    `DEPLOYMENT_ID=${metadata.timestamp}`,
    `GIT_COMMIT_SHA=${metadata.commit}`,
  ];
  
  const command = `vercel --prod ${envVars.map(v => `--env ${v}`).join(' ')}`;
  
  const output = runCommand(command);
  
  // Extract deployment URL
  const urlMatch = output.match(/https:\/\/[^\s]+/);
  const deploymentUrl = urlMatch ? urlMatch[0] : null;
  
  return deploymentUrl;
}

// Health check
async function performHealthCheck(url: string) {
  console.log(`\n${colors.bright}🏥 Performing health check...${colors.reset}\n`);
  
  const healthUrl = `${url}/api/health?detailed=true`;
  let attempts = 0;
  
  while (attempts < config.checks.health.retries) {
    attempts++;
    console.log(`Attempt ${attempts}/${config.checks.health.retries}...`);
    
    try {
      const response = await fetch(healthUrl, {
        signal: AbortSignal.timeout(config.checks.health.timeout),
      });
      
      const data = await response.json();
      
      if (data.status === 'healthy') {
        console.log(`${colors.green}✓ Health check passed${colors.reset}`);
        return true;
      }
      
      console.log(`${colors.yellow}⚠ Health check returned:${colors.reset} ${data.status}`);
      
    } catch (error) {
      console.error(`${colors.red}✗ Health check failed:${colors.reset} ${error}`);
    }
    
    // Wait before retry
    if (attempts < config.checks.health.retries) {
      await new Promise(resolve => setTimeout(resolve, 10000));
    }
  }
  
  return false;
}

// Send deployment notification
async function sendNotification(status: 'success' | 'failure', metadata: any, details?: any) {
  const message = {
    text: `Deployment ${status === 'success' ? '✅ Succeeded' : '❌ Failed'}`,
    blocks: [
      {
        type: 'section',
        text: {
          type: 'mrkdwn',
          text: `*Deployment ${status === 'success' ? 'Succeeded' : 'Failed'}*\n` +
                `Version: ${metadata.version}\n` +
                `Commit: ${metadata.commit}\n` +
                `Deployer: ${metadata.deployer}\n` +
                `Time: ${metadata.timestamp}`,
        },
      },
    ],
  };
  
  if (details) {
    message.blocks.push({
      type: 'section',
      text: {
        type: 'mrkdwn',
        text: `*Details:*\n${JSON.stringify(details, null, 2)}`,
      },
    });
  }
  
  // Send to Slack
  if (config.notifications.slack) {
    try {
      await fetch(config.notifications.slack, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(message),
      });
    } catch (error) {
      console.error('Failed to send Slack notification:', error);
    }
  }
}

// Rollback deployment
async function rollbackDeployment(previousTag: string) {
  console.log(`\n${colors.bright}⏮ Rolling back deployment...${colors.reset}\n`);
  
  const response = await prompts({
    type: 'confirm',
    name: 'confirm',
    message: `Roll back to ${previousTag}?`,
    initial: false,
  });
  
  if (response.confirm) {
    runCommand(`vercel rollback ${previousTag} --yes`);
    console.log(`${colors.green}✓ Rolled back to ${previousTag}${colors.reset}`);
  }
}

// Main deployment flow
async function deploy() {
  console.log(`${colors.bright}🚀 Production Deployment Script${colors.reset}\n`);
  
  let deploymentTag: string | null = null;
  let deploymentUrl: string | null = null;
  
  try {
    // Check prerequisites
    if (!await checkPrerequisites()) {
      throw new Error('Prerequisites check failed');
    }
    
    // Run validation
    if (!await runValidationChecks()) {
      throw new Error('Validation checks failed');
    }
    
    // Run checklist
    if (!await runChecklist()) {
      throw new Error('Deployment checklist incomplete');
    }
    
    // Get metadata
    const metadata = getDeploymentMetadata();
    
    // Create backup
    console.log(`\n${colors.bright}💾 Creating database backup...${colors.reset}`);
    runCommand('npm run backup:database');
    
    // Create deployment tag
    deploymentTag = await createDeploymentTag(metadata);
    console.log(`${colors.green}✓ Created tag: ${deploymentTag}${colors.reset}`);
    
    // Deploy
    deploymentUrl = await deployToVercel(metadata);
    
    if (!deploymentUrl) {
      throw new Error('Failed to get deployment URL');
    }
    
    console.log(`${colors.green}✓ Deployed to: ${deploymentUrl}${colors.reset}`);
    
    // Health check
    const healthy = await performHealthCheck(deploymentUrl);
    
    if (!healthy) {
      throw new Error('Health check failed');
    }
    
    // Success notification
    await sendNotification('success', metadata, { url: deploymentUrl });
    
    console.log(`\n${colors.green}${colors.bright}✅ Deployment successful!${colors.reset}\n`);
    
    // Save deployment info
    const deploymentInfo = {
      ...metadata,
      url: deploymentUrl,
      tag: deploymentTag,
      status: 'success',
    };
    
    writeFileSync(
      join(process.cwd(), 'deployment-log.json'),
      JSON.stringify(deploymentInfo, null, 2)
    );
    
  } catch (error) {
    console.error(`\n${colors.red}${colors.bright}❌ Deployment failed!${colors.reset}\n`);
    console.error(error);
    
    // Failure notification
    await sendNotification('failure', getDeploymentMetadata(), { error: String(error) });
    
    // Offer rollback
    if (deploymentTag) {
      const previousTag = runCommand('git describe --tags --abbrev=0 HEAD^', true).trim();
      await rollbackDeployment(previousTag);
    }
    
    process.exit(1);
  }
}

// Run if called directly
if (require.main === module) {
  deploy();
}

// Export for use in other scripts
export { deploy, checkPrerequisites, runValidationChecks };
