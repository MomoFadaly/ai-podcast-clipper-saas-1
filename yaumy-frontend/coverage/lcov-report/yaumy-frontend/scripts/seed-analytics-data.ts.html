#!/usr/bin/env tsx

import { PrismaClient } from '@prisma/client';

const db = new PrismaClient();

async function seedAnalyticsData() {
  console.log('🌱 Seeding analytics data...');

  const now = new Date();
  const yesterday = new Date(now.getTime() - 24 * 60 * 60 * 1000);
  const weekAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000);

  // Get existing users
  const users = await db.user.findMany({
    select: { id: true },
    take: 10,
  });
  
  const userIds = users.map(u => u.id);
  console.log(`Found ${userIds.length} users in database`);

  // Create sample sessions
  const sessions = [];
  for (let i = 0; i < 50; i++) {
    const sessionDate = new Date(now.getTime() - Math.random() * 7 * 24 * 60 * 60 * 1000);
    const sessionId = `sess_${Math.random().toString(36).substr(2, 9)}${Date.now().toString(36)}`;
    
    const browsers = ['Chrome', 'Safari', 'Firefox', 'Edge'];
    const devices = ['desktop', 'mobile', 'tablet'];
    const countries = ['United States', 'United Kingdom', 'Canada', 'Germany', 'France', 'Australia'];
    const referrers = ['https://google.com', 'https://facebook.com', 'https://twitter.com', '', 'https://linkedin.com'];
    const utmSources = ['google-ads', 'social-media', 'email-campaign', 'organic', null];
    
    const session = await db.analyticsSession.create({
      data: {
        sessionId,
        userId: Math.random() > 0.3 && userIds.length > 0 ? userIds[Math.floor(Math.random() * userIds.length)] : null,
        startedAt: sessionDate,
        endedAt: new Date(sessionDate.getTime() + Math.random() * 60 * 60 * 1000), // Random session length up to 1 hour
        entryPage: Math.random() > 0.5 ? '/' : ['/dashboard', '/dashboard/library', '/login', '/pricing'][Math.floor(Math.random() * 4)],
        exitPage: Math.random() > 0.5 ? '/' : ['/dashboard', '/dashboard/library', '/login', '/pricing'][Math.floor(Math.random() * 4)],
        entryReferrer: referrers[Math.floor(Math.random() * referrers.length)],
        pageViews: Math.floor(Math.random() * 10) + 1,
        userAgent: `Mozilla/5.0 (compatible; ${browsers[Math.floor(Math.random() * browsers.length)]})`,
        browser: browsers[Math.floor(Math.random() * browsers.length)],
        os: ['Windows', 'macOS', 'Linux', 'iOS', 'Android'][Math.floor(Math.random() * 5)],
        device: devices[Math.floor(Math.random() * devices.length)],
        isMobile: Math.random() > 0.7,
        country: countries[Math.floor(Math.random() * countries.length)],
        region: ['California', 'London', 'Ontario', 'Bavaria', 'Île-de-France', 'New South Wales'][Math.floor(Math.random() * 6)],
        city: ['San Francisco', 'London', 'Toronto', 'Munich', 'Paris', 'Sydney'][Math.floor(Math.random() * 6)],
        timezone: 'America/Los_Angeles',
        utmSource: utmSources[Math.floor(Math.random() * utmSources.length)],
        utmMedium: Math.random() > 0.5 ? ['cpc', 'social', 'email', 'organic'][Math.floor(Math.random() * 4)] : null,
        utmCampaign: Math.random() > 0.7 ? 'summer-2024' : null,
      },
    });
    
    sessions.push(session);
  }

  // Create page views
  const pages = ['/', '/dashboard', '/dashboard/library', '/dashboard/explore', '/login', '/pricing', '/about'];
  for (const session of sessions) {
    const pageViewCount = Math.floor(Math.random() * 5) + 1;
    for (let i = 0; i < pageViewCount; i++) {
      await db.analyticsPageView.create({
        data: {
          userId: session.userId,
          sessionId: session.sessionId,
          page: pages[Math.floor(Math.random() * pages.length)],
          title: `Page ${Math.floor(Math.random() * 100)}`,
          referrer: i === 0 ? session.entryReferrer : null,
          loadTime: Math.floor(Math.random() * 3000) + 500, // 500ms to 3.5s
          timestamp: new Date(session.startedAt.getTime() + i * 5 * 60 * 1000), // 5 minutes apart
        },
      });
    }
  }

  // Create events
  const events = ['click_button', 'form_submit', 'video_play', 'download', 'search', 'share'];
  for (let i = 0; i < 100; i++) {
    const session = sessions[Math.floor(Math.random() * sessions.length)];
    await db.analyticsEvent.create({
      data: {
        eventName: events[Math.floor(Math.random() * events.length)],
        userId: session.userId,
        sessionId: session.sessionId,
        page: pages[Math.floor(Math.random() * pages.length)],
        referrer: session.entryReferrer,
        userAgent: session.userAgent,
        browser: session.browser,
        os: session.os,
        device: session.device,
        isMobile: session.isMobile,
        country: session.country,
        region: session.region,
        city: session.city,
        timezone: session.timezone,
        utmSource: session.utmSource,
        utmMedium: session.utmMedium,
        utmCampaign: session.utmCampaign,
        properties: {
          random_property: Math.random() > 0.5 ? 'value' : 'other_value',
          numeric_property: Math.floor(Math.random() * 100),
        },
        timestamp: new Date(session.startedAt.getTime() + Math.random() * 60 * 60 * 1000),
      },
    });
  }

  console.log('✅ Analytics data seeded successfully!');
  console.log(`Created ${sessions.length} sessions with page views and events`);
}

seedAnalyticsData()
  .catch((e) => {
    console.error(e);
    process.exit(1);
  })
  .finally(async () => {
    await db.$disconnect();
  });
