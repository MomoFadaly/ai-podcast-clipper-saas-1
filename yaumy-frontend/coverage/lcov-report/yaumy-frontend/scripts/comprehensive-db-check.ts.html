interface TableInfo {
  table_name: string;
}

interface ColumnInfo {
  column_name: string;
  data_type: string;
  is_nullable: string;
  column_default: string | null;
}

interface ConstraintInfo {
  table_name: string;
  constraint_name: string;
  constraint_type: string;
}

interface IndexInfo {
  tablename: string;
  indexname: string;
}

      'notes_v2', 'note_blocks_v2', 'block_types', 'note_shares_v2',

      'LayoutPreset', 'UserLayoutPreference', 'LayoutUsageAnalytics', 'SavedLayout',
      'PodcastCache', 'ai_summaries', '_prisma_migrations'
    ];
    
    const tables = await prisma.$queryRaw<TableInfo[]>`
      SELECT table_name 
      FROM information_schema.tables 
      WHERE table_schema = 'public' 
      AND table_type = 'BASE TABLE'
      ORDER BY table_name;
    `;
    
    const existingTables = tables.map(t => t.table_name);
    
    for (const table of expectedTables) {
      const exists = existingTables.includes(table);
      console.log(`${exists ? '✅' : '❌'} ${table}`);
    }
    
    const missingTables = expectedTables.filter(t => !existingTables.includes(t));

    // 2. Check User table columns specifically
    console.log('\n\n📋 CHECKING USER TABLE COLUMNS:');
    console.log('--------------------------------');
    
    const userColumns = await prisma.$queryRaw<ColumnInfo[]>`
      SELECT column_name, data_type, is_nullable, column_default
      FROM information_schema.columns
      WHERE table_name = 'User'
      ORDER BY column_name;
    `;
    
    const requiredUserColumns = [
      'id', 'name', 'email', 'emailVerified', 'password',
      'stripeCustomerId', 'subscriptionPlan', 'subscriptionStatus',
      'subscriptionEndDate', 'stripeSubscriptionId', 'image'

    const existingUserColumns = userColumns.map(c => c.column_name);
    
    for (const col of requiredUserColumns) {
      const exists = existingUserColumns.includes(col);

    }
    
    // 3. Check critical table columns
    console.log('\n\n🔍 CHECKING CRITICAL TABLE STRUCTURES:');

    // Check UploadedFile columns
    const uploadedFileColumns = await prisma.$queryRaw<ColumnInfo[]>`
      SELECT column_name FROM information_schema.columns
      WHERE table_name = 'UploadedFile'
      AND column_name IN ('processingProgress', 'sourceType', 'listenNotesEpisodeId', 'podcastTitle');
    `;
    
    console.log('\nUploadedFile table:');
    console.log(`  ${uploadedFileColumns.find(c => c.column_name === 'processingProgress') ? '✅' : '❌'} processingProgress`);
    console.log(`  ${uploadedFileColumns.find(c => c.column_name === 'sourceType') ? '✅' : '❌'} sourceType`);

    const constraints = await prisma.$queryRaw<ConstraintInfo[]>`
      SELECT tc.table_name, tc.constraint_name, tc.constraint_type
      FROM information_schema.table_constraints tc
      WHERE tc.table_schema = 'public'
      AND tc.constraint_type = 'FOREIGN KEY'
      ORDER BY tc.table_name;
    `;
    
    const constraintsByTable: Record<string, number> = {};
    constraints.forEach(c => {
      constraintsByTable[c.table_name] = (constraintsByTable[c.table_name] || 0) + 1;
    });
    
    console.log(`Total foreign key constraints: ${constraints.length}`);
    console.log('\nConstraints per table:');
    Object.entries(constraintsByTable).forEach(([table, count]) => {

    const indexes = await prisma.$queryRaw<IndexInfo[]>`

      WHERE schemaname = 'public'
      AND indexname NOT LIKE '%_pkey'
      ORDER BY tablename, indexname;
    `;
    
    const indexesByTable: Record<string, number> = {};
    indexes.forEach(idx => {

    });

    Object.entries(indexesByTable).forEach(([table, count]) => {
      console.log(`  ${table}: ${count} indexes`);
