    },
    {
      id: "resource-2",
      title: "Advanced React Patterns",
      type: "video" as const,
      url: "https://youtu.be/MlDTHzK1vKI",
      creator: "Kent C. Dodds",
      duration: "45 min video",
      description: "Advanced React patterns and techniques",
      difficulty: "advanced" as const,
      isFree: true,
    },
    {
      id: "resource-3",
      title: "React Best Practices",
      type: "article" as const,
      url: "https://medium.com/react-best-practices",
      creator: "Medium Author",
      duration: "10 min read",
      description: "Best practices for React development",
      difficulty: "intermediate" as const,
      isFree: true,
    },
    {
      id: "resource-4",
      title: "React Tutorial",
      type: "video" as const,
      url: "youtube.com/watch?v=dGcsHMXbSOA", // Missing https://
      creator: "Dev Ed",
      duration: "30 min video",
      description: "React tutorial for beginners",
      difficulty: "beginner" as const,
      isFree: true,
    },
  ];

  console.log("Total resources:", testResources.length);
  console.log("Resources:", testResources.map(r => ({ title: r.title, type: r.type, url: r.url })));

  // Filter YouTube resources - same logic as in saveLearningPathAsTrack
  const youtubeResources = testResources.filter(
    (r) => r.type === "video" && r.url && (r.url.includes("youtube.com") || r.url.includes("youtu.be"))
  );

  console.log("\nFiltered YouTube resources:", youtubeResources.length);
  console.log("YouTube resources:", youtubeResources.map(r => ({ title: r.title, url: r.url })));

  // Check each resource
  console.log("\nDetailed analysis:");
  testResources.forEach((resource, index) => {
    console.log(`\nResource ${index + 1}: ${resource.title}`);
    console.log(`  Type: ${resource.type}`);
    console.log(`  URL: ${resource.url}`);
    console.log(`  Has URL: ${!!resource.url}`);
    console.log(`  Is video type: ${resource.type === "video"}`);
    console.log(`  Contains youtube.com: ${resource.url?.includes("youtube.com")}`);
    console.log(`  Contains youtu.be: ${resource.url?.includes("youtu.be")}`);
    console.log(`  Would be filtered: ${resource.type === "video" && resource.url && (resource.url.includes("youtube.com") || resource.url.includes("youtu.be"))}`);
  });

  // Test URL extraction
  console.log("\n\nTesting URL extraction:");
  youtubeResources.forEach((resource) => {
    console.log(`\nResource: ${resource.title}`);
    console.log(`  Original URL: ${resource.url}`);
    
    // Extract video ID (basic extraction)
    let videoId = null;
    if (resource.url.includes("youtube.com/watch?v=")) {
      videoId = resource.url.split("v=")[1]?.split("&")[0];
    } else if (resource.url.includes("youtu.be/")) {
      videoId = resource.url.split("youtu.be/")[1]?.split("?")[0];
    }
    
    console.log(`  Extracted video ID: ${videoId}`);
  });

  console.log("\n\nDone!");
  process.exit(0);
}

testYouTubeResourceFiltering().catch((error) => {
  console.error("Error:", error);
  process.exit(1);
});
