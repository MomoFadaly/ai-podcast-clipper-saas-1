/**
 * Simple Adobe Layout Context
 * 
 * Global state management for the Adobe layout system
 * 
 * ⚠️ CRITICAL ARCHITECTURAL DECISIONS:
 * 
 * 1. GLOBAL drag state (not local component state)
 *    - Required for cross-panel drag operations
 *    - Local state would isolate drags to single panels
 *    - DO NOT move dragOperation to component state
 * 
 * 2. Panel references by ID (not object references)
 *    - Object refs become stale during drag/resize operations
 *    - IDs remain stable across re-renders
 *    - Prevents "undefined" errors during state updates
 * 
 * 3. Immer for state mutations with error boundaries
 *    - Complex nested state requires careful mutation
 *    - safeImmerProduce wrapper prevents crashes
 *    - DO NOT bypass the wrapper
 * 
 * 4. Validation functions for JSON operations
 *    - Preset data comes from database as JSON
 *    - Invalid JSON structure would crash the app
 *    - Validation ensures type safety
 * 
 * See ADOBE_LAYOUT_CRITICAL_DOCS.md for detailed explanations
 */

// CRITICAL: Type validation functions to prevent JSON parsing crashes
const isValidString = (value: unknown): value is string => {
  return typeof value === 'string';
};
 
const isValidNumber = (value: unknown): value is number => {
  return typeof value === 'number' && !isNaN(value);
};
 
const isValidArray = (value: unknown): value is unknown[] => {
  return Array.isArray(value);
};
 
const isValidObject = (value: unknown): value is Record<string, unknown> => {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
};
 
const validateDockZone = (data: unknown): data is DockZone => {
  if (!isValidObject(data)) return false;
  
  const zone = data;
  
  // Required fields
  if (!isValidString(zone.id)) {
    return false;
  }
  
  if (zone.orientation !== 'horizontal' && zone.orientation !== 'vertical') {
    return false;
  }
  
  if (!isValidArray(zone.children)) {
    return false;
  }
  
  // Optional size validation
  if (zone.size !== undefined && !isValidNumber(zone.size)) {
    return false;
  }
  
  // Recursively validate children
  for (const child of zone.children) {
    if (isValidObject(child) && 'type' in child && child.type === 'panel-group') {
      if (!validatePanelGroup(child)) return false;
    } else {
      if (!validateDockZone(child)) return false;
    }
  }
  
  return true;
};
 
const validatePanelGroup = (data: unknown): data is PanelGroup => {
  if (!isValidObject(data)) return false;
  
  const group = data;
  
  // Required fields
  if (!isValidString(group.id)) {
    return false;
  }
  
  if (group.type !== 'panel-group') {
    return false;
  }
  
  if (!isValidArray(group.tabs)) {
    return false;
  }
  
  // Validate all tabs are strings
  for (const tab of group.tabs) {
    if (!isValidString(tab)) {
      return false;
    }
  }
  
  if (!isValidString(group.activeTabId)) {
    return false;
  }
  
  // Optional size validation
  if (group.size !== undefined && !isValidNumber(group.size)) {
    return false;
  }
  
  return true;
};
 
const validatePanel = (data: unknown): data is Panel => {
  if (!isValidObject(data)) return false;
  
  const panel = data;
  
  // Required fields
  if (!isValidString(panel.id)) {
    return false;
  }
  
  if (!isValidString(panel.title)) {
    return false;
  }
  
  // content can be any React node, so we just check it exists
  if (panel.content === undefined) {
    return false;
  }
  
  return true;
};
 
// eslint-disable-next-line @typescript-eslint/no-unused-vars
const validatePresetData = (data: unknown): { isValid: boolean; rootZone?: DockZone; panelAssignments?: Record<string, string[]> } => {
  if (!isValidObject(data)) {
    return { isValid: false };
  }
  
  const preset = data;
  
  // Validate rootZone
  if (!preset.rootZone || !validateDockZone(preset.rootZone)) {
    return { isValid: false };
  }
  
  // Validate panelAssignments (optional)
  let panelAssignments: Record<string, string[]> | undefined;
  if (preset.panelAssignments !== undefined) {
    if (!isValidObject(preset.panelAssignments)) {
      return { isValid: false };
    }
    
    const assignments = preset.panelAssignments;
    panelAssignments = {};
    
    for (const [groupId, tabs] of Object.entries(assignments)) {
      if (!isValidString(groupId)) {
        return { isValid: false };
      }
      
      if (!isValidArray(tabs)) {
        return { isValid: false };
      }
      
      const validTabs: string[] = [];
      for (const tab of tabs) {
        if (!isValidString(tab)) {
          return { isValid: false };
        }
        validTabs.push(tab);
      }
      
      panelAssignments[groupId] = validTabs;
    }
  }
  
  return { 
    isValid: true, 
    rootZone: preset.rootZone, 
    panelAssignments 
  };
};

  jsonString: string,

      return null;

      const grandParentInfo = findParent(parent.id, rootZone);
      
      if (grandParentInfo && remainingChild) {
        remainingChild.size = parent.size;
        grandParentInfo.parent.children[grandParentInfo.index] = remainingChild;
      }
    } else if (parent.children.length > 1) {
      // Redistribute sizes among remaining children
      const avgSize = 100 / parent.children.length;
      parent.children.forEach(child => {
        child.size = avgSize;
      });
    }
  }, [findParent]);
 
  // Toggle maximize/minimize for a panel
  const toggleMaximizePanel = useCallback((panelId: string) => {
    setState(prev => {
      if (prev.maximizedPanelId === panelId) {
        // Restore from maximized state
        if (prev.preMaximizeLayout) {
          return {
            ...prev,
            rootZone: prev.preMaximizeLayout,
            maximizedPanelId: null,
            preMaximizeLayout: null
          };
        }
        return prev;
      } else {
        // Maximize the panel
        const panel = prev.panels.get(panelId);
        if (!panel) return prev;
        
        // Create a single panel layout
        const maximizedLayout: DockZone = {
          id: 'root',
          orientation: 'horizontal',
          children: [{
            id: 'maximized-group',
            type: 'panel-group',
            tabs: [panelId],
            activeTabId: panelId,
            size: 100
          } as PanelGroup]
        };
        
        return {
          ...prev,
          rootZone: maximizedLayout,
          maximizedPanelId: panelId,
          preMaximizeLayout: prev.rootZone
        };
      }
    });
  }, []);
 
  // Initialize layout from saved state
  const initializeLayout = useCallback(async () => {
    
    // Check if layout has been manually modified (has non-default sizes)
    const hasCustomSizes = (zone: DockZone): boolean => {
      if (zone.children) {
        for (const child of zone.children) {
          if ('size' in child && child.size !== undefined) {
            // Check if this zone has non-default sizes
            if (child.id === 'main-content' && child.size !== 75) return true;
            if (child.id === 'sidebar' && child.size !== 25) return true;
            if (child.id === 'video-group' && child.size !== 70) return true;
            if (child.id === 'timeline-controls' && child.size !== 30) return true;
            if (child.id === 'sidebar-top' && child.size !== 50) return true;
            if (child.id === 'sidebar-bottom' && child.size !== 50) return true;
          }
          if (!('type' in child) && hasCustomSizes(child)) {
            return true;
          }
        }
      }
      return false;
    };
    
    const defaultLayout = await persistence.loadDefaultPreset();
    if (defaultLayout) {
      setState(prev => {
        // If the current layout has custom sizes, don&apos;t override it
        if (hasCustomSizes(prev.rootZone)) {
          return prev;
        }
        
        const newRootZone = defaultLayout.rootZone;
        
        // Find all panel groups in the default layout
        const findAllPanelGroups = (zone: DockZone): PanelGroup[] => {
          const groups: PanelGroup[] = [];
          for (const child of zone.children) {
            if ('type' in child && child.type === 'panel-group') {
              groups.push(child);
            } else if (!('type' in child)) {
              groups.push(...findAllPanelGroups(child));
            }
          }
          return groups;
        };
        
        const allGroups = findAllPanelGroups(newRootZone);
        
        // If we have existing panels, redistribute them
        if (prev.panels.size > 0) {
          const panelIds = Array.from(prev.panels.keys());
          
          // Clear all tabs from groups first
          allGroups.forEach(group => {
            group.tabs = [];
            group.activeTabId = '';
          });
          
          // Distribute panels to groups
          panelIds.forEach((panelId, index) => {
            if (allGroups.length > 0) {
              const targetGroup = allGroups[index % allGroups.length];
              if (targetGroup) {
                targetGroup.tabs.push(panelId);
                if (targetGroup.activeTabId === '') {
                  targetGroup.activeTabId = panelId;
                }
              }
            }
          });
        }
        
        return {
          ...prev,
          rootZone: newRootZone,
          panels: prev.panels
        };
      });
    }
  }, [persistence]);
 
  const contextValue: SimpleAdobeLayoutContextValue = useMemo(() => ({
    state,
    setState,
    addPanel,
    addPanels,
    setActiveTab,
    removePanel,
    splitPanelGroup,
    collapseEmptyGroup,
    toggleMaximizePanel,
    saveAsPreset,
    loadPreset,
    availablePresets: persistence.availablePresets,
    currentPreset: persistence.currentPreset,
    isSaving: persistence.isSaving,
    isLoading: persistence.isLoading,
    clearCurrentPreset: persistence.clearCurrentPreset,
    initializeLayout,
    isInitialized
  }), [
    state, 
    setState, 
    addPanel, 
    addPanels, 
    setActiveTab,
    removePanel, 
    splitPanelGroup,
    collapseEmptyGroup,
    toggleMaximizePanel,
    saveAsPreset, 
    loadPreset,
    persistence.availablePresets,
    persistence.currentPreset,
    persistence.isSaving,
    persistence.isLoading,
    persistence.clearCurrentPreset,
    initializeLayout,
    isInitialized
  ]);
 
  return (
    <SimpleAdobeLayoutContext.Provider value={contextValue}>
      {children}
    </SimpleAdobeLayoutContext.Provider>
  );
}
 
export function useSimpleAdobeLayout() {
  const context = useContext(SimpleAdobeLayoutContext);
  if (!context) {
    throw new Error('useSimpleAdobeLayout must be used within SimpleAdobeLayoutProvider');
  }
  return context;
}
 
export function useSimpleAdobeLayoutSafe() {
  const context = useContext(SimpleAdobeLayoutContext);
  return context; // Returns null if not within provider
}
