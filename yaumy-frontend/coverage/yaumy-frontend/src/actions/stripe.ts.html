"use server";

import { redirect } from "next/navigation";
import Stripe from "stripe";
import { env } from "~/env";
import { auth } from "~/server/auth";
import { db } from "~/server/db";

const stripe = new Stripe(env.STRIPE_SECRET_KEY, {
  apiVersion: "2025-05-28.basil",
});

export type PlanId = "core" | "power";

// You'll need to create these price IDs in your Stripe dashboard
const PRICE_IDS: Record<string, string> = {
  power_monthly: env.STRIPE_POWER_MONTHLY_PRICE ?? "price_power_monthly",
  power_yearly: env.STRIPE_POWER_YEARLY_PRICE ?? "price_power_yearly",
};

export async function getUserSubscription() {
  const serverSession = await auth();
  if (!serverSession?.user?.id) return null;

  const user = await db.user.findUnique({
    where: { id: serverSession.user.id },
    select: { 
      subscriptionPlan: true,
      subscriptionStatus: true,
      subscriptionEndDate: true,
      stripeCustomerId: true,
      stripeSubscriptionId: true,
    }
  });

  if (!user?.subscriptionPlan || user.subscriptionStatus !== 'active') {
    return { plan: 'core' };
  }

  return {
    plan: user.subscriptionPlan as PlanId,
    status: user.subscriptionStatus,
    endDate: user.subscriptionEndDate,
  };
}

export async function createCheckoutSession(planId: PlanId, interval: 'monthly' | 'yearly' = 'monthly') {
  if (planId === 'core') {
    redirect('/dashboard');
    return;
  }

  const serverSession = await auth();
  if (!serverSession?.user?.id) {
    redirect('/login');
    return;
  }

  const user = await db.user.findUniqueOrThrow({
    where: { id: serverSession.user.id },
    select: { 
      stripeCustomerId: true,
      email: true,
    },
  });

  let customerId = user.stripeCustomerId;

  // Create Stripe customer if doesn't exist
  if (!customerId) {
    const customer = await stripe.customers.create({
      email: user.email,
      metadata: {
        userId: serverSession.user.id,
      },
    });
    
    customerId = customer.id;
    
    await db.user.update({
      where: { id: serverSession.user.id },
      data: { stripeCustomerId: customerId },
    });
  }

  const priceId = interval === 'yearly' ? PRICE_IDS.power_yearly : PRICE_IDS.power_monthly;

  const session = await stripe.checkout.sessions.create({
    line_items: [{ price: priceId, quantity: 1 }],
    customer: customerId,
    mode: "subscription",
    success_url: `${env.BASE_URL}/dashboard/billing?success=true`,
    cancel_url: `${env.BASE_URL}/dashboard/billing`,
    metadata: {
      userId: serverSession.user.id,
      planId: planId,
    },
  });

  if (!session.url) {
    throw new Error("Failed to create session URL");
  }

  redirect(session.url);
}

export async function cancelSubscription() {
  const serverSession = await auth();
  if (!serverSession?.user?.id) {
    redirect('/login');
    return;
  }

  const user = await db.user.findUniqueOrThrow({
    where: { id: serverSession.user.id },
    select: { stripeSubscriptionId: true },
  });

  if (!user.stripeSubscriptionId) {
    throw new Error("No active subscription found");
  }

  // Cancel at period end
  await stripe.subscriptions.update(user.stripeSubscriptionId, {
    cancel_at_period_end: true,
  });

  // Update user record
  await db.user.update({
    where: { id: serverSession.user.id },
    data: { 
      subscriptionStatus: 'canceling',
    },
  });

  redirect('/dashboard/billing?canceled=true');
}
