"use server";

import { GetObjectCommand, S3Client } from "@aws-sdk/client-s3";
import { getSignedUrl } from "@aws-sdk/s3-request-presigner";
import { revalidatePath } from "next/cache";
import { env } from "~/env";
import { inngest } from "~/inngest/client";
import { auth } from "~/server/auth";
import { db } from "~/server/db";

export async function processVideo(
  uploadedFileId: string,
  chunks: Array<{ start: number; end: number }>,
  fileInfo?: {
    fileCategory?: "video" | "audio" | "document";
    fileName?: string;
    contentType?: string;
    chunkingMethod?: "minutes" | "chunks" | "pages" | "words";
    chunkingValue?: number;
    fileSize?: number;
  },
) {
  const uploadedFile = await db.uploadedFile.findUniqueOrThrow({
    where: {
      id: uploadedFileId,
    },
    select: {
      uploaded: true,
      id: true,
      userId: true,
      s3Key: true,
      displayName: true,
      contentType: true,
    },
  });

  if (uploadedFile.uploaded) return;

  // Determine file category
  const fileCategory = fileInfo?.fileCategory || (() => {
    const contentType = fileInfo?.contentType || uploadedFile.contentType || "";
    const fileName = fileInfo?.fileName || uploadedFile.displayName || "";
    const extension = fileName.toLowerCase().split('.').pop() || '';
    
    if (contentType.startsWith("video/")) return "video";
    
    if (contentType.startsWith("audio/") || 
        ["mp3", "wav", "m4a", "aac", "ogg", "flac", "wma", "opus"].includes(extension)) {
      return "audio";
    }
    
    return "document";
  })();

  const isAudioFile = fileCategory === "audio";
  const isDocumentFile = fileCategory === "document";

  await inngest.send({
    name: "process-video-events",
    data: {
      uploadedFileId: uploadedFile.id,
      userId: uploadedFile.userId,
      s3Key: uploadedFile.s3Key,
      chunks,
      // Add file-specific fields
      isAudioFile,
      isDocumentFile,
      fileCategory,
      fileName: fileInfo?.fileName || uploadedFile.displayName,
      contentType: fileInfo?.contentType || uploadedFile.contentType,
      // Add chunking configuration
      chunkingMethod: fileInfo?.chunkingMethod,
      chunkingValue: fileInfo?.chunkingValue,
      // Add file size for S3 tracking
      fileSize: fileInfo?.fileSize,
    },
  });

  await db.uploadedFile.update({
    where: {
      id: uploadedFileId,
    },
    data: {
      uploaded: true,
    },
  });

  revalidatePath("/dashboard");
}

export async function getClipPlayUrl(
  clipId: string,
): Promise<{ success: boolean; url?: string; error?: string }> {
  const session = await auth();
  if (!session?.user?.id) {
    return { success: false, error: "Unauthorized" };
  }

  try {
    const clip = await db.clip.findUniqueOrThrow({
      where: {
        id: clipId,
        userId: session.user.id,
      },
    });

    const s3Client = new S3Client({
      region: env.AWS_REGION,
      credentials: {
        accessKeyId: env.AWS_ACCESS_KEY_ID,
        secretAccessKey: env.AWS_SECRET_ACCESS_KEY,
      },
    });

    const command = new GetObjectCommand({
      Bucket: env.S3_BUCKET_NAME,
      Key: clip.s3Key,
    });

    const signedUrl = await getSignedUrl(s3Client, command, {
      expiresIn: 3600,
    });

    return { success: true, url: signedUrl };
  } catch (error) {
    console.error("Error generating clip play URL:", error);
    return { success: false, error: "Failed to generate play URL." };
  }
}
