/**
 * Adobe Panel Group Component
 * 
 * CRITICAL COMPONENT - Contains essential drag/drop functionality
 * 
 * ⚠️ WARNING: This component contains several patterns that may seem unnecessary
 * but are REQUIRED for proper functionality:
 * 
 * 1. setTimeout(0) in drag handlers - REQUIRED for visual drag feedback
 * 2. Global state for drag operations - REQUIRED for cross-panel dragging
 * 3. No memoization on TabComponent - REQUIRED for drag state updates
 * 4. Complex drop handling - REQUIRED for proper panel splitting
 * 
 * See ADOBE_LAYOUT_CRITICAL_DOCS.md for detailed explanations
 */
'use client';

import React, { useState, useRef, useMemo, useCallback, useEffect } from 'react';
import { useSimpleAdobeLayout } from '~/contexts/simple-adobe-layout-context';
import { useKeyboardNavigation } from '~/hooks/use-keyboard-navigation';
import type { PanelGroup, Panel } from '~/types/adobe-layout';
import { cn } from '~/lib/utils';
import { Plus, MoreVertical, ArrowRight, ArrowDown, XCircle } from 'lucide-react';
import {
  DropdownMenu,
  DropdownMenuContent,
  DropdownMenuItem,
  DropdownMenuTrigger,
} from '~/components/ui/dropdown-menu';
import { Button } from '~/components/ui/button';
// EmptyPanel removed - using custom implementation

interface AdobePanelGroupProps {
  group: PanelGroup;
}

/**
 * TabComponent - Draggable tab UI element
 * 
 * ⚠️ CRITICAL: This component is NOT memoized on purpose!
 * Memoization would prevent drag state updates from triggering re-renders,
 * breaking the visual feedback during drag operations.
 * 
 * DO NOT add React.memo() to this component!
 */
const TabComponent = ({ 
  panel, 
  index, 
  isActive, 
  isDragged, 
  _dragOverIndex, 
  _isInternalDrag,
  isDragActive, 
  onDragStart, 
  onDragEnd, 
  onDragOver,
  onDragEnter,
  onDragLeave,
  onDrop, 
  onTabClick, 
  onCloseTab: _onCloseTab,
  groupId
}: {
  panel: Panel;
  index: number;
  isActive: boolean;
  isDragged: boolean;
  _dragOverIndex: number | null;
  _isInternalDrag: boolean;
  isDragActive: boolean;
  onDragStart: (e: React.DragEvent) => void;
  onDragEnd: (e: React.DragEvent) => void;
  onDragOver: (e: React.DragEvent) => void;
  onDragEnter: (e: React.DragEvent) => void;
  onDragLeave: (e: React.DragEvent) => void;
  onDrop: (e: React.DragEvent) => void;
  onTabClick: () => void;
  onCloseTab: (e: React.MouseEvent) => void;
  groupId: string;
}) => {
  const tabId = `tab-${groupId}-${panel.id}`;
  const panelId = `panel-${groupId}-${panel.id}`;


  return (
    <div
      id={tabId}
      role="tab"
      tabIndex={isActive ? 0 : -1}
      aria-selected={isActive}
      aria-controls={panelId}
      aria-label={`${panel.title} panel tab`}
      aria-describedby={`tab-description-${panel.id}`}
      draggable={true}
      onDragStart={onDragStart}
      onDragEnd={onDragEnd}
      onDragOver={onDragOver}
      onDragEnter={onDragEnter}
      onDragLeave={onDragLeave}
      onDrop={onDrop}
      onClick={(e) => {
        // Prevent click during drag operations
        if (!isDragged && !isDragActive) {
          onTabClick();
        } else {
          e.preventDefault();
          e.stopPropagation();
        }
      }}
      onKeyDown={(e) => {
        if (e.key === 'Enter' || e.key === ' ') {
          e.preventDefault();
          onTabClick();
        }
      }}
      className="px-3 py-2 bg-blue-100 border border-blue-300 m-1 cursor-move"
      style={{
        cursor: isDragActive ? 'grabbing' : 'grab',
        userSelect: 'none'
      }}
    >
      <span>
        📄 {panel.title}
      </span>
      
      
      {/* Hidden description for screen readers */}
      <span 
        id={`tab-description-${panel.id}`}
        className="sr-only"
      >
        Panel {index + 1} of {/* This will be filled by parent */}. 
        {isActive ? 'Currently active' : 'Click to activate'}.
        Press Tab to navigate, Space or Enter to select.
      </span>
    </div>
  );
};

TabComponent.displayName = 'TabComponent';

export const AdobePanelGroup = React.memo(({ group }: AdobePanelGroupProps) => {
  const { state, setActiveTab, setState, splitPanelGroup, collapseEmptyGroup, removePanel } = useSimpleAdobeLayout();
  const { currentFocus } = useKeyboardNavigation();
  const [isCollapsed, setIsCollapsed] = useState(false);
  // Simple local state for visual feedback only
  const [isDraggedOver, setIsDraggedOver] = useState(false);
  
  
  const groupRef = useRef<HTMLDivElement>(null);
  
  // Check if this group is currently focused for accessibility
  const isFocused = currentFocus?.groupId === group.id;

  // Get panels for this group
  const panels = useMemo(() => {
    // Ensure state.panels is a Map
    const panelsMap = state.panels instanceof Map ? state.panels : new Map();
    const groupPanels = group.tabs
      .map(tabId => panelsMap.get(tabId))
      .filter((panel): panel is Panel => panel !== undefined);
    
    // Debug logging to track panel availability
    if (group.tabs.length > 0 && groupPanels.length === 0) {
      console.warn(`[PanelGroup ${group.id}] No panels found for tabs:`, group.tabs);
      console.warn(`[PanelGroup ${group.id}] Available panels:`, Array.from(panelsMap.keys()));
    }
    
    return groupPanels;
  }, [group.tabs, state.panels, group.id]);

  // Handlers
  const handleCloseTab = useCallback((_panelId: string) => {
    // TODO: Implement tab closing functionality
  }, []);


  /**
   * createTabDragHandlers - Factory for drag event handlers
   * 
   * ⚠️ CRITICAL IMPLEMENTATION DETAILS:
   * 
   * 1. The setTimeout(0) in onDragStart is ABSOLUTELY NECESSARY
   *    - Without it, React's state update blocks the browser's drag ghost creation
   *    - This causes the tab to not visually move with the cursor
   *    - This is a known React/Browser interaction issue
   * 
   * 2. Global state updates (setState) are REQUIRED
   *    - Local state would only work within a single panel
   *    - Global state enables dragging between different panels
   * 
   * 3. The dataTransfer.setData calls must happen BEFORE setTimeout
   *    - Browser security requires synchronous data setting in drag start
   * 
   * DO NOT "OPTIMIZE" THIS CODE - IT WILL BREAK!
   */
  const createTabDragHandlers = useCallback((panel: Panel, index: number) => {
    return {
      onDragStart: (e: React.DragEvent) => {
        console.log('🔥 TAB DRAG START!', panel.title);
        
        // Essential drag setup - ONLY browser-native operations
        e.dataTransfer.effectAllowed = 'move';
        e.dataTransfer.setData('text/plain', panel.id);
        e.dataTransfer.setData('application/panel-id', panel.id);
        e.dataTransfer.setData('application/source-group', group.id);
        
        // CRITICAL: setTimeout(0) is REQUIRED for drag and drop to work visually
        // React state updates in onDragStart prevent the browser from creating
        // the drag ghost image. This defers the state update until after the
        // browser has captured the drag element for the ghost.
        // DO NOT REMOVE - this will break visual drag feedback
        setTimeout(() => {
          setState(prev => ({
            ...prev,
            dragOperation: {
              panel,
              sourceGroupId: group.id,
              sourceIndex: index,
              timestamp: Date.now()
            }
          }));
        }, 0);
      },
      
      onDragEnd: (_e: React.DragEvent) => {
        console.log('🔥 TAB DRAG END!', panel.title);
        
        // Clear all drag state
        setState(prev => ({
          ...prev,
          dragOperation: null
        }));
        
        // Clear visual feedback
        setIsDraggedOver(false);
      },
      
      onDragOver: (e: React.DragEvent) => {
        e.preventDefault();
        e.stopPropagation();
      },
      
      onDragEnter: (e: React.DragEvent) => {
        e.preventDefault();
        e.stopPropagation();
      },
      
      onDragLeave: (e: React.DragEvent) => {
        e.preventDefault();
        e.stopPropagation();
      },
      
      onDrop: (e: React.DragEvent) => {
        e.preventDefault();
        e.stopPropagation();
      },
      
      onTabClick: () => {
        setActiveTab(group.id, panel.id);
      },
      
      onCloseTab: (e: React.MouseEvent) => {
        e.stopPropagation();
        handleCloseTab(panel.id);
      }
    };
  }, [group.id, setState, setActiveTab, handleCloseTab]);

  // CRITICAL: Cleanup drag state on unmount and visibility changes to prevent memory leaks
  useEffect(() => {
    const handleVisibilityChange = () => {
      if (document.visibilityState === 'hidden' && state.dragOperation) {
        setState(prev => ({ ...prev, dragOperation: null }));
        setIsDraggedOver(false);
      }
    };

    document.addEventListener('visibilitychange', handleVisibilityChange);
    return () => document.removeEventListener('visibilitychange', handleVisibilityChange);
  }, [state.dragOperation, setState]);

  // Complete drop operation - handles both tab addition and panel splitting - RACE CONDITION FIXED
  useCallback((dropInfo: { zoneId: string; position: string; rect: DOMRect }) => {
    // Use current drag state from global state
    const currentDragState = state.dragOperation;
    
    if (!currentDragState?.panel) {
      console.warn('[DROP] No active drag operation found');
      return;
    }
    
    console.log('[DROP] Using synchronous drag state:', currentDragState);
    
    setState(prev => {
      const newRootZone = JSON.parse(JSON.stringify(prev.rootZone));
      const panel = currentDragState.panel;
      const sourceGroupId = currentDragState.sourceGroupId;
      
      // Helper function to find a zone or group by ID
      const findById = (id: string, zone: any): any => {
        if (zone.id === id) return zone;
        for (const child of zone.children) {
          if ('type' in child && child.type === 'panel-group' && child.id === id) {
            return child;
          } else if (!('type' in child)) {
            const found = findById(id, child);
            if (found) return found;
          }
        }
        return null;
      };

      // Helper function to find parent
      const findParent = (id: string, zone: any): { parent: any; index: number } | null => {
        for (let i = 0; i < zone.children.length; i++) {
          const child = zone.children[i];
          if (child.id === id) {
            return { parent: zone, index: i };
          }
          if (!('type' in child)) {
            const found = findParent(id, child);
            if (found) return found;
          }
        }
        return null;
      };

      // Remove panel from source group
      const sourceGroup = findById(sourceGroupId, newRootZone);
      if (sourceGroup && sourceGroup.type === 'panel-group') {
        sourceGroup.tabs = sourceGroup.tabs.filter((tabId: string) => tabId !== panel.id);
        
        // Update active tab if needed
        if (sourceGroup.activeTabId === panel.id) {
          sourceGroup.activeTabId = sourceGroup.tabs.length > 0 ? sourceGroup.tabs[0] : '';
        }
        
        // Handle empty source group cleanup
        if (sourceGroup.tabs.length === 0 && !sourceGroup.id.startsWith('empty-group-')) {
          collapseEmptyGroup(newRootZone, sourceGroupId);
        }
      }
      
      if (dropInfo.position === 'center') {
        // Add as tab to existing group
        const targetGroup = findById(dropInfo.zoneId, newRootZone);
        if (targetGroup && targetGroup.type === 'panel-group') {
          if (!targetGroup.tabs.includes(panel.id)) {
            targetGroup.tabs.push(panel.id);
            targetGroup.activeTabId = panel.id;
          }
        }
      } else {
        // Create split layout
        const target = findById(dropInfo.zoneId, newRootZone);
        if (!target) return prev;

        const parentInfo = findParent(dropInfo.zoneId, newRootZone);
        if (!parentInfo) return prev;

        const { parent, index } = parentInfo;
        const position = dropInfo.position as 'top' | 'bottom' | 'left' | 'right';
        const isVerticalSplit = position === 'top' || position === 'bottom';
        const newOrientation = isVerticalSplit ? 'vertical' : 'horizontal';
        
        // Create new panel group
        const newGroup = {
          id: `group-${Date.now()}`,
          type: 'panel-group',
          tabs: [panel.id],
          activeTabId: panel.id,
          size: 50
        };

        // Check if we need to create a container or can insert directly
        if (parent.orientation === newOrientation) {
          // Same orientation - insert directly
          const insertIndex = position === 'top' || position === 'left' ? index : index + 1;
          parent.children.splice(insertIndex, 0, newGroup);
          
          // Redistribute sizes equally
          const avgSize = 100 / parent.children.length;
          parent.children.forEach((child: any) => {
            child.size = avgSize;
          });
        } else {
          // Different orientation - create wrapper container
          const targetCopy = JSON.parse(JSON.stringify(target));
          const children = position === 'top' || position === 'left' 
            ? [newGroup, targetCopy] 
            : [targetCopy, newGroup];
          
          children.forEach((child: any) => {
            child.size = 50;
          });

          const newContainer = {
            id: `zone-${Date.now()}`,
            orientation: newOrientation,
            children,
            size: target.size
          };

          parent.children[index] = newContainer;
        }
      }
      
      return {
        ...prev,
        rootZone: newRootZone,
        dragOperation: null
      };
    });
  }, [state.dragOperation, setState, collapseEmptyGroup]);

  // Get active panel
  const activePanel = panels.find(p => p.id === group.activeTabId);

  return (
    <div 
      ref={groupRef}
      role="tabpanel"
      aria-label={`Panel group ${group.id}`}
      aria-describedby={`group-description-${group.id}`}
      className={cn(
        "adobe-panel-group h-full w-full flex flex-col bg-white border border-gray-200 rounded-lg shadow-sm overflow-hidden relative transition-all duration-200",
        isFocused && "ring-2 ring-blue-500 ring-opacity-50",
        isDraggedOver && "ring-2 ring-green-400 bg-green-50"
      )}
      onDragOver={(e) => {
        e.preventDefault();
        
        // Only show visual feedback if dragging from different group
        const sourceGroupId = e.dataTransfer.getData('application/source-group');
        if (sourceGroupId && sourceGroupId !== group.id) {
          setIsDraggedOver(true);
        }
      }}
      onDragLeave={(e) => {
        // Clear visual feedback when leaving
        if (!e.currentTarget.contains(e.relatedTarget as Node)) {
          setIsDraggedOver(false);
        }
      }}
      onDrop={(e) => {
        e.preventDefault();
        e.stopPropagation();
        
        const panelId = e.dataTransfer.getData('application/panel-id');
        const sourceGroupId = e.dataTransfer.getData('application/source-group');
        
        console.log('✅ DROP:', { panelId, sourceGroupId, targetGroup: group.id });
        
        // Clear visual feedback
        setIsDraggedOver(false);
        
        // Move panel if from different group
        if (panelId && sourceGroupId && sourceGroupId !== group.id) {
          setState(prev => {
            const newRootZone = JSON.parse(JSON.stringify(prev.rootZone));
            
            // Helper to find and update panel groups
            const updatePanelGroups = (zone: any) => {
              for (const child of zone.children) {
                if (child.type === 'panel-group') {
                  if (child.id === group.id) {
                    // Add to target group
                    if (!child.tabs.includes(panelId)) {
                      child.tabs.push(panelId);
                      child.activeTabId = panelId;
                    }
                  } else if (child.id === sourceGroupId) {
                    // Remove from source group
                    child.tabs = child.tabs.filter((id: string) => id !== panelId);
                    if (child.activeTabId === panelId) {
                      child.activeTabId = child.tabs[0] || '';
                    }
                  }
                } else if (!child.type) {
                  updatePanelGroups(child);
                }
              }
            };
            
            updatePanelGroups(newRootZone);
            
            return {
              ...prev,
              rootZone: newRootZone,
              dragOperation: null
            };
          });
        }
      }}
    >
      {/* Hidden description for screen readers */}
      <span 
        id={`group-description-${group.id}`}
        className="sr-only"
      >
        Panel group with {panels.length} panels. 
        Use Tab to navigate between panels, Ctrl+Shift+H to split horizontally, 
        Ctrl+Shift+V to split vertically.
      </span>
      {/* Header */}
      <div 
        className={cn(
          "flex items-center justify-between border-b border-gray-200 bg-gray-50 relative z-10",
          isCollapsed ? "px-3 py-2" : "px-2 py-1"
        )}
      >
        {/* Tabs or Collapsed Header */}
        {isCollapsed ? (
          <div className="flex items-center gap-2 flex-1">
            <button
              onClick={() => setIsCollapsed(false)}
              className="hover:bg-gray-200 p-1 rounded transition-colors duration-150"
            >
              <ArrowRight className="h-4 w-4" />
            </button>
            <span className="text-sm font-medium text-gray-700">{activePanel?.title || 'Empty'}</span>
          </div>
        ) : (
          <>
            <div 
              role="tablist"
              aria-label={`Tabs for ${group.id} panel group`}
              aria-describedby={`tablist-description-${group.id}`}
              className="flex items-center gap-1 flex-1 overflow-x-auto"
              onDragOver={(e) => {
                // Prevent interference with child drag operations
                e.stopPropagation();
              }}
            >
              {/* Hidden description for screen readers */}
              <span 
                id={`tablist-description-${group.id}`}
                className="sr-only"
              >
                Tab list with {panels.length} tabs. Use arrow keys to navigate, Enter or Space to activate.
              </span>
              
              {panels.map((panel, index) => {
                const tabHandlers = createTabDragHandlers(panel, index);
                return (
                  <TabComponent
                    key={panel.id}
                    panel={panel}
                    index={index}
                    isActive={panel.id === group.activeTabId}
                    isDragged={state.dragOperation?.panel?.id === panel.id}
                    _dragOverIndex={null}
                    _isInternalDrag={false}
                    isDragActive={!!state.dragOperation}
                    groupId={group.id}
                    {...tabHandlers}
                  />
                );
              })}
              
              {/* Drop indicator when dragging - using global state */}
              {state.dragOperation && state.dragOperation.sourceGroupId !== group.id && (
                <div className="flex items-center px-3 py-1 text-xs text-blue-600 bg-blue-50 border-2 border-dashed border-blue-300 rounded-md animate-pulse">
                  <div className="w-3 h-3 mr-1 bg-blue-500 rounded-full flex items-center justify-center">
                    <Plus className="w-2 h-2 text-white" />
                  </div>
                  <span className="font-medium">Drop here to add tab</span>
                </div>
              )}
            </div>

            {/* Actions */}
            <div className="flex items-center gap-1 px-2">
              <Button
                variant="ghost"
                size="icon"
                className="h-6 w-6"
                onClick={() => setIsCollapsed(true)}
              >
                <ArrowDown className="h-3 w-3" />
              </Button>
              <DropdownMenu>
                <DropdownMenuTrigger asChild>
                  <Button variant="ghost" size="icon" className="h-6 w-6">
                    <MoreVertical className="h-3 w-3" />
                  </Button>
                </DropdownMenuTrigger>
                <DropdownMenuContent align="end" className="w-48">
                  <DropdownMenuItem onClick={() => splitPanelGroup(group.id, 'right')} disabled={panels.length === 0}>
                    <ArrowRight className="mr-2 h-4 w-4" />
                    Split Right
                  </DropdownMenuItem>
                  <DropdownMenuItem onClick={() => splitPanelGroup(group.id, 'bottom')} disabled={panels.length === 0}>
                    <ArrowDown className="mr-2 h-4 w-4" />
                    Split Down
                  </DropdownMenuItem>
                  <DropdownMenuItem>
                    <Plus className="mr-2 h-4 w-4" />
                    Add Panel
                  </DropdownMenuItem>
                  <DropdownMenuItem 
                    disabled={panels.length === 0}
                    onClick={() => {
                      // Remove all panels in this group
                      panels.forEach(panel => {
                        removePanel(panel.id);
                      });
                    }}
                  >
                    <XCircle className="mr-2 h-4 w-4" />
                    Close All
                  </DropdownMenuItem>
                </DropdownMenuContent>
              </DropdownMenu>
            </div>
          </>
        )}
      </div>

      {/* Content */}
      {!isCollapsed && (
        <div 
          className="flex-1 min-h-0 relative"
          role="tabpanel"
          aria-labelledby={activePanel ? `tab-${group.id}-${activePanel.id}` : undefined}
          aria-live="polite"
          aria-atomic="true"
        >
          {activePanel ? (
            <div 
              id={`panel-${group.id}-${activePanel.id}`}
              className="h-full w-full overflow-auto panel-content-scrollable"
              aria-label={`${activePanel.title} panel content`}
            >
              {activePanel.content}
            </div>
          ) : (
            <div 
              className="h-full w-full flex items-center justify-center bg-gray-50"
              role="status"
              aria-label="Empty panel group"
            >
              <div className="text-center">
                <div className="mb-4 text-gray-400">
                  <Plus className="h-12 w-12 mx-auto" aria-hidden="true" />
                </div>
                <p className="text-gray-600 mb-2">No panels in this group</p>
                <Button
                  variant="outline"
                  size="sm"
                  onClick={() => {/* TODO: Implement add panel */}}
                  aria-label="Add new panel to this group"
                >
                  <Plus className="h-4 w-4 mr-2" aria-hidden="true" />
                  Add Panel
                </Button>
              </div>
            </div>
          )}
        </div>
      )}

      {/* Simple visual feedback when being dragged over */}
      {isDraggedOver && (
        <div className="absolute inset-4 border-2 border-dashed border-green-400 bg-green-50/20 rounded-lg flex items-center justify-center z-20 pointer-events-none">
          <div className="bg-white px-4 py-2 rounded-lg shadow-lg border border-green-400">
            <span className="text-green-700 font-medium">Drop here to add tab</span>
          </div>
        </div>
      )}
    </div>
  );
});
