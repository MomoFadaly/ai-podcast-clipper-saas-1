'use client';

import { useCallback } from 'react';
import { SafeEmptyPanel } from './safe-empty-panel';
import { useSimpleAdobeLayout } from '~/contexts/simple-adobe-layout-context';
import type { PanelGroup } from '~/types/adobe-layout';
import { 
  Video,
  FileText,
  StickyNote,
  Clock,
  Navigation,
  List,
  Film,
  BarChart3
} from 'lucide-react';

interface EmptyPanelContainerProps {
  group: PanelGroup;
  contentType?: 'VIDEO' | 'AUDIO' | 'TEXT' | 'PDF';
  availableData?: {
    hasTranscript?: boolean;
    hasMultipleChunks?: boolean;
  };
  panelFactory?: (panelId: string) => React.ReactNode;
}

// Panel definitions
const PANEL_DEFINITIONS = {
  'media-player': { title: 'Media Player', icon: Video },
  'text-reader': { title: 'Text Reader', icon: FileText },
  'transcript': { title: 'Transcript', icon: FileText },
  'notes': { title: 'Notes', icon: StickyNote },
  'timeline': { title: 'Timeline', icon: Clock },
  'navigation': { title: 'Navigation', icon: Navigation },
  'chunk-navigation-vertical': { title: 'Chapters', icon: List },
  'chunk-navigation-horizontal': { title: 'Horizontal Navigation', icon: Film },
  'progress': { title: 'Progress', icon: BarChart3 },
  'ai-summary': { title: 'AI Summary', icon: BarChart3 }, // Will need Sparkles import
};

/**
 * Container component that handles safe panel creation for empty panels
 */
export function EmptyPanelContainer({
  group,
  contentType = 'VIDEO',
  availableData = {},
  panelFactory
}: EmptyPanelContainerProps) {
  const { addPanel, clearCurrentPreset } = useSimpleAdobeLayout();

  // Handle panel addition safely
  const handlePanelAdded = useCallback((panelId: string) => {
    console.log('[EmptyPanelContainer] Adding panel:', panelId);
    
    // First check if the panel factory can create this panel
    const panelContent = panelFactory ? panelFactory(panelId) : null;
    
    if (!panelContent) {
      console.warn('[EmptyPanelContainer] Panel not available from factory:', panelId);
      
      // Show a warning to the user
      const panelDef = PANEL_DEFINITIONS[panelId as keyof typeof PANEL_DEFINITIONS];
      const title = panelDef?.title || panelId;
      
      // For now, create a placeholder that explains why the panel isn't available
      const IconComponent = panelDef?.icon || BarChart3;
      const placeholderPanel = {
        id: `${panelId}-unavailable-${Date.now()}`,
        title: `${title} (Unavailable)`,
        icon: <IconComponent className="w-3 h-3 opacity-50" />,
        content: (
          <div className="h-full w-full flex items-center justify-center bg-gray-50">
            <div className="text-center p-8 max-w-md">
              <IconComponent className="w-16 h-16 text-gray-300 mx-auto mb-4" />
              <h3 className="text-lg font-medium text-gray-700 mb-2">{title}</h3>
              <p className="text-sm text-gray-500 mb-4">
                This panel is not available for the current content
              </p>
              {panelId === 'timeline' && (
                <p className="text-xs text-amber-600">
                  Timeline requires multiple chunks. This content may only have a single chunk.
                </p>
              )}
              {(panelId === 'navigation' || panelId.includes('chunk-navigation')) && (
                <p className="text-xs text-amber-600">
                  Navigation panels require multiple chunks to be useful.
                </p>
              )}
              {panelId === 'transcript' && (
                <p className="text-xs text-amber-600">
                  Transcript may not be available for this content type.
                </p>
              )}
            </div>
          </div>
        ),
        closable: true
      };
      
      // Add the placeholder panel temporarily
      clearCurrentPreset();
      addPanel(placeholderPanel, group.id);
      return;
    }
    
    const panelDef = PANEL_DEFINITIONS[panelId as keyof typeof PANEL_DEFINITIONS];
    if (!panelDef) {
      console.error('[EmptyPanelContainer] Unknown panel type:', panelId);
      return;
    }

    try {
      // Create the panel with the actual content from the factory
      const IconComponent = panelDef.icon;
      const panel = {
        id: panelId,
        title: panelDef.title,
        icon: <IconComponent className="w-3 h-3" />,
        content: panelContent,
        closable: true
      };

      // Clear current preset to allow saving
      clearCurrentPreset();

      // Add the panel
      addPanel(panel, group.id);
      
      console.log('[EmptyPanelContainer] Panel added successfully with factory content');
    } catch (error) {
      console.error('[EmptyPanelContainer] Error adding panel:', error);
    }
  }, [addPanel, clearCurrentPreset, panelFactory, group.id]);

  return (
    <SafeEmptyPanel
      group={group}
      contentType={contentType}
      hasMultipleChunks={availableData.hasMultipleChunks}
      onPanelAdded={handlePanelAdded}
    />
  );
}
