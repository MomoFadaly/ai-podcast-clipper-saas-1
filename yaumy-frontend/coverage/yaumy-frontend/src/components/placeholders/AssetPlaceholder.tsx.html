'use client'

import { cn } from '~/lib/utils'

interface AssetPlaceholderProps {
  name: string
  description: string
  width?: number | string
  height?: number | string
  className?: string
  children?: React.ReactNode
}

export function AssetPlaceholder({
  name,
  description,
  width = 200,
  height = 200,
  className = '',
  children
}: AssetPlaceholderProps) {
  return (
    <div 
      className={cn(
        "border-2 border-dashed border-gray-400 bg-gray-100 dark:bg-gray-800 dark:border-gray-600 rounded-lg flex flex-col items-center justify-center p-4 text-center",
        className
      )}
      style={{ width, height }}
    >
      <div className="text-sm font-mono text-gray-600 dark:text-gray-400 mb-2">
        ASSET_PLACEHOLDER
      </div>
      <div className="text-lg font-semibold text-gray-800 dark:text-gray-200 mb-1">
        {name}
      </div>
      <div className="text-xs text-gray-500 dark:text-gray-500 max-w-full">
        {description}
      </div>
      {children && (
        <div className="mt-2 w-full">
          {children}
        </div>
      )}
    </div>
  )
}
