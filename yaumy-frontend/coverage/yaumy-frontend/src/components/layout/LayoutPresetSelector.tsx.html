"use client";

import React, { useState, useRef, useEffect } from 'react';
import { 
  Monitor, 
  ChevronDown,
  Save, 
  RotateCcw, 
  Plus,
  Check,
  Clock,
  Loader2
} from 'lucide-react';
import { Button } from '~/components/ui/button';
import { cn } from '~/lib/utils';

interface LayoutPreset {
  id: string;
  name: string;
  description?: string | null;
  category: string;
  isPublic: boolean;
  usageCount: number;
}

interface LayoutPresetSelectorProps {
  currentPreset: LayoutPreset | null;
  availablePresets: LayoutPreset[];
  hasUnsavedChanges: boolean;
  isSaving: boolean;
  lastSaved: Date | null;
  onPresetSelect: (presetId: string) => Promise<void>;
  onSaveCurrentLayout: (name?: string) => Promise<void>;
  onSaveAsPreset: (name: string, description?: string) => Promise<LayoutPreset>;
  onResetToDefault: () => Promise<void>;
  className?: string;
}

export function LayoutPresetSelector({
  currentPreset,
  availablePresets,
  hasUnsavedChanges,
  isSaving,
  lastSaved,
  onPresetSelect,
  onSaveCurrentLayout,
  onSaveAsPreset,
  onResetToDefault,
  className
}: LayoutPresetSelectorProps) {
  const [isOpen, setIsOpen] = useState(false);
  const [showSaveDialog, setShowSaveDialog] = useState(false);
  const [newPresetName, setNewPresetName] = useState('');
  const [newPresetDescription, setNewPresetDescription] = useState('');
  const [isCreatingPreset, setIsCreatingPreset] = useState(false);
  const dropdownRef = useRef<HTMLDivElement>(null);

  // Close dropdown when clicking outside
  useEffect(() => {
    function handleClickOutside(event: MouseEvent) {
      if (dropdownRef.current && !dropdownRef.current.contains(event.target as Node)) {
        setIsOpen(false);
        setShowSaveDialog(false);
      }
    }
    document.addEventListener("mousedown", handleClickOutside);
    return () => document.removeEventListener("mousedown", handleClickOutside);
  }, []);

  const handlePresetSelect = async (presetId: string) => {
    try {
      await onPresetSelect(presetId);
      setIsOpen(false);
    } catch (error) {
      console.error('Failed to switch preset:', error);
    }
  };

  const handleSaveAsPreset = async () => {
    if (!newPresetName.trim()) return;
    
    try {
      setIsCreatingPreset(true);
      await onSaveAsPreset(newPresetName.trim(), newPresetDescription.trim() || undefined);
      setNewPresetName('');
      setNewPresetDescription('');
      setShowSaveDialog(false);
      setIsOpen(false);
    } catch (error) {
      console.error('Failed to save preset:', error);
    } finally {
      setIsCreatingPreset(false);
    }
  };

  const handleQuickSave = async () => {
    try {
      await onSaveCurrentLayout();
    } catch (error) {
      console.error('Failed to save layout:', error);
    }
  };

  const handleReset = async () => {
    try {
      await onResetToDefault();
      setIsOpen(false);
    } catch (error) {
      console.error('Failed to reset layout:', error);
    }
  };

  // Group presets by category
  const presetsByCategory = availablePresets.reduce((acc, preset) => {
    const category = preset.category || 'Custom';
    if (!acc[category]) {
      acc[category] = [];
    }
    acc[category].push(preset);
    return acc;
  }, {} as Record<string, LayoutPreset[]>);

  const formatLastSaved = (date: Date) => {
    const now = new Date();
    const diffMs = now.getTime() - date.getTime();
    const diffMins = Math.floor(diffMs / 60000);
    
    if (diffMins < 1) return 'Just now';
    if (diffMins < 60) return `${diffMins}m ago`;
    const diffHours = Math.floor(diffMins / 60);
    if (diffHours < 24) return `${diffHours}h ago`;
    return date.toLocaleDateString();
  };

  return (
    <div className={cn("relative", className)} ref={dropdownRef}>
      <div className="flex items-center space-x-2">
        {/* Save Status Indicator */}
        {(hasUnsavedChanges || isSaving) && (
          <div className="flex items-center space-x-1 text-xs text-gray-500">
            {isSaving ? (
              <>
                <Loader2 className="h-3 w-3 animate-spin" />
                <span>Saving...</span>
              </>
            ) : (
              <>
                <div className="h-2 w-2 bg-orange-400 rounded-full" />
                <span>Unsaved</span>
              </>
            )}
          </div>
        )}

        {/* Quick Save Button */}
        {hasUnsavedChanges && !isSaving && (
          <Button
            variant="ghost"
            size="sm"
            onClick={handleQuickSave}
            className="h-8 px-2 text-xs hover:bg-gray-100"
          >
            <Save className="h-3 w-3 mr-1" />
            Save
          </Button>
        )}

        {/* Layout Preset Selector */}
        <Button
          variant="outline"
          size="sm"
          onClick={() => setIsOpen(!isOpen)}
          className={cn(
            "flex items-center space-x-2 h-8 px-3 text-xs border-gray-300 hover:bg-gray-50",
            isOpen && "bg-gray-50"
          )}
        >
          <Monitor className="h-3 w-3" />
          <span className="max-w-24 truncate">
            {currentPreset?.name || 'Default Layout'}
          </span>
          <ChevronDown className={cn(
            "h-3 w-3 transition-transform duration-200",
            isOpen ? "rotate-180" : ""
          )} />
        </Button>
      </div>

      {/* Dropdown Menu */}
      {isOpen && (
        <div className="absolute right-0 mt-2 w-80 bg-white rounded-lg shadow-lg border border-gray-200 py-2 z-50">
          {/* Header */}
          <div className="px-4 py-2 border-b border-gray-100">
            <div className="flex items-center justify-between">
              <h3 className="text-sm font-semibold text-gray-900">Layout Presets</h3>
              {lastSaved && (
                <div className="flex items-center space-x-1 text-xs text-gray-500">
                  <Clock className="h-3 w-3" />
                  <span>{formatLastSaved(lastSaved)}</span>
                </div>
              )}
            </div>
          </div>

          {/* Current Layout Actions */}
          <div className="px-2 py-2 border-b border-gray-100">
            <div className="flex items-center space-x-2">
              <Button
                variant="ghost"
                size="sm"
                onClick={() => setShowSaveDialog(true)}
                className="flex-1 justify-start h-8 text-xs"
              >
                <Plus className="h-3 w-3 mr-2" />
                Save as New Preset
              </Button>
              <Button
                variant="ghost"
                size="sm"
                onClick={handleReset}
                className="h-8 px-2 text-xs text-gray-600 hover:text-gray-900"
                title="Reset to Default"
              >
                <RotateCcw className="h-3 w-3" />
              </Button>
            </div>
          </div>

          {/* Save Dialog */}
          {showSaveDialog && (
            <div className="px-4 py-3 bg-gray-50 border-b border-gray-200">
              <div className="space-y-3">
                <div>
                  <label className="block text-xs font-medium text-gray-700 mb-1">
                    Preset Name
                  </label>
                  <input
                    type="text"
                    value={newPresetName}
                    onChange={(e) => setNewPresetName(e.target.value)}
                    placeholder="My Custom Layout"
                    className="w-full px-2 py-1 text-xs border border-gray-300 rounded focus:outline-none focus:ring-1 focus:ring-blue-500"
                    autoFocus
                  />
                </div>
                <div>
                  <label className="block text-xs font-medium text-gray-700 mb-1">
                    Description (optional)
                  </label>
                  <input
                    type="text"
                    value={newPresetDescription}
                    onChange={(e) => setNewPresetDescription(e.target.value)}
                    placeholder="Optimized for video editing"
                    className="w-full px-2 py-1 text-xs border border-gray-300 rounded focus:outline-none focus:ring-1 focus:ring-blue-500"
                  />
                </div>
                <div className="flex items-center space-x-2">
                  <Button
                    variant="default"
                    size="sm"
                    onClick={handleSaveAsPreset}
                    disabled={!newPresetName.trim() || isCreatingPreset}
                    className="flex-1 h-7 text-xs"
                  >
                    {isCreatingPreset ? (
                      <>
                        <Loader2 className="h-3 w-3 mr-1 animate-spin" />
                        Saving...
                      </>
                    ) : (
                      <>
                        <Save className="h-3 w-3 mr-1" />
                        Save Preset
                      </>
                    )}
                  </Button>
                  <Button
                    variant="ghost"
                    size="sm"
                    onClick={() => setShowSaveDialog(false)}
                    className="h-7 px-2 text-xs"
                  >
                    Cancel
                  </Button>
                </div>
              </div>
            </div>
          )}

          {/* Preset List */}
          <div className="max-h-64 overflow-y-auto">
            {Object.entries(presetsByCategory).map(([category, presets]) => (
              <div key={category}>
                <div className="px-4 py-2">
                  <h4 className="text-xs font-medium text-gray-500 uppercase tracking-wide">
                    {category}
                  </h4>
                </div>
                <div className="px-2">
                  {presets.map((preset) => (
                    <button
                      key={preset.id}
                      onClick={() => handlePresetSelect(preset.id)}
                      className={cn(
                        "w-full flex items-center justify-between px-2 py-2 rounded text-left hover:bg-gray-100 transition-colors duration-150",
                        currentPreset?.id === preset.id && "bg-blue-50 hover:bg-blue-100"
                      )}
                    >
                      <div className="flex-1 min-w-0">
                        <div className="flex items-center space-x-2">
                          <span className={cn(
                            "text-sm font-medium truncate",
                            currentPreset?.id === preset.id ? "text-blue-900" : "text-gray-900"
                          )}>
                            {preset.name}
                          </span>
                          {currentPreset?.id === preset.id && (
                            <Check className="h-3 w-3 text-blue-600 flex-shrink-0" />
                          )}
                        </div>
                        {preset.description && (
                          <p className="text-xs text-gray-500 truncate mt-0.5">
                            {preset.description}
                          </p>
                        )}
                      </div>
                      {preset.usageCount > 0 && (
                        <span className="text-xs text-gray-400 ml-2">
                          {preset.usageCount}
                        </span>
                      )}
                    </button>
                  ))}
                </div>
              </div>
            ))}
          </div>

          {/* Empty State */}
          {availablePresets.length === 0 && (
            <div className="px-4 py-8 text-center">
              <Monitor className="h-8 w-8 text-gray-400 mx-auto mb-2" />
              <p className="text-sm text-gray-500">No layout presets available</p>
              <p className="text-xs text-gray-400">Create your first preset above</p>
            </div>
          )}
        </div>
      )}
    </div>
  );
}
