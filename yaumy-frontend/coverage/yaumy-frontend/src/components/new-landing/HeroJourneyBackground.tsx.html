'use client'

import { useEffect, useRef } from 'react'
import { gsap } from 'gsap'
import { ScrollTrigger } from 'gsap/ScrollTrigger'

if (typeof window !== 'undefined') {
  gsap.registerPlugin(ScrollTrigger)
}

export function HeroJourneyBackground() {
  const bgRef = useRef<HTMLDivElement>(null)
  
  useEffect(() => {
    if (!bgRef.current) return
    
    // Animate background opacity based on scroll
    const tl = gsap.timeline({
      scrollTrigger: {
        trigger: document.body,
        start: 'top top',
        end: 'bottom bottom',
        scrub: 1,
      }
    })
    
    // Gradually reveal lighter layers as user scrolls
    tl.to('.bg-layer-1', { opacity: 0.8 }, 0)
      .to('.bg-layer-2', { opacity: 0.85 }, 0.1)
      .to('.bg-layer-3', { opacity: 0.9 }, 0.2)
      .to('.bg-layer-4', { opacity: 0.95 }, 0.3)
      .to('.bg-layer-5', { opacity: 1 }, 0.4)
      .to('.bg-layer-6', { opacity: 1 }, 0.5)
      .to('.bg-layer-7', { opacity: 1 }, 0.6)
      .to('.bg-layer-8', { opacity: 1 }, 0.7)
      .to('.bg-layer-9', { opacity: 1 }, 0.8)
      .to('.bg-layer-10', { opacity: 1 }, 0.9)
    
    return () => {
      ScrollTrigger.getAll().forEach(trigger => trigger.kill())
    }
  }, [])
  
  return (
    <div ref={bgRef} className="fixed inset-0 -z-10">
      {/* Base gradient matching the provided image */}
      <div className="absolute inset-0 bg-gradient-to-b from-[#0a1628] via-[#0d1f3c] via-[#112950] via-[#1a3a5c] via-[#234b68] via-[#2d5a7b] via-[#3a6f8f] via-[#4a90a4] via-[#5ba3b6] via-[#67b7c4] to-[#87ceeb]" />
      
      {/* Additional overlay for smooth transitions */}
      <div className="bg-layer-1 absolute inset-0 opacity-0 bg-gradient-to-b from-[#0a1628]/90 to-[#0d1f3c]/85" />
      <div className="bg-layer-2 absolute inset-0 opacity-0 bg-gradient-to-b from-[#0d1f3c]/85 to-[#112950]/80" />
      <div className="bg-layer-3 absolute inset-0 opacity-0 bg-gradient-to-b from-[#112950]/80 to-[#1a3a5c]/75" />
      <div className="bg-layer-4 absolute inset-0 opacity-0 bg-gradient-to-b from-[#1a3a5c]/75 to-[#234b68]/70" />
      <div className="bg-layer-5 absolute inset-0 opacity-0 bg-gradient-to-b from-[#234b68]/70 to-[#2d5a7b]/65" />
      <div className="bg-layer-6 absolute inset-0 opacity-0 bg-gradient-to-b from-[#2d5a7b]/65 to-[#3a6f8f]/60" />
      <div className="bg-layer-7 absolute inset-0 opacity-0 bg-gradient-to-b from-[#3a6f8f]/60 to-[#4a90a4]/55" />
      <div className="bg-layer-8 absolute inset-0 opacity-0 bg-gradient-to-b from-[#4a90a4]/55 to-[#5ba3b6]/50" />
      <div className="bg-layer-9 absolute inset-0 opacity-0 bg-gradient-to-b from-[#5ba3b6]/50 to-[#67b7c4]/45" />
      <div className="bg-layer-10 absolute inset-0 opacity-0 bg-gradient-to-b from-[#67b7c4]/45 to-[#87ceeb]/40" />
      
      {/* Animated elements */}
      <ParticleField />
      <FloatingClouds />
      <LightRays />
    </div>
  )
}

function ParticleField() {
  const particlesRef = useRef<HTMLDivElement>(null)
  
  useEffect(() => {
    if (!particlesRef.current) return
    
    // Create floating particles that fade out as user scrolls
    const particles = Array.from({ length: 50 }, (_, i) => {
      const particle = document.createElement('div')
      particle.className = 'particle absolute w-1 h-1 bg-white rounded-full'
      particle.style.left = `${Math.random() * 100}%`
      particle.style.top = `${Math.random() * 100}%`
      particlesRef.current?.appendChild(particle)
      return particle
    })
    
    // Animate particles
    particles.forEach((particle, i) => {
      gsap.to(particle, {
        y: -window.innerHeight,
        opacity: 0,
        duration: 10 + Math.random() * 20,
        repeat: -1,
        delay: Math.random() * 10,
        ease: 'none',
      })
    })
    
    // Fade out particle field as user scrolls
    gsap.to(particlesRef.current, {
      opacity: 0,
      scrollTrigger: {
        trigger: document.body,
        start: 'top top',
        end: '50% top',
        scrub: 1,
      }
    })
    
    return () => {
      particles.forEach(p => p.remove())
    }
  }, [])
  
  return <div ref={particlesRef} className="absolute inset-0 opacity-50" />
}

function FloatingClouds() {
  const cloudsRef = useRef<HTMLDivElement>(null)
  
  useEffect(() => {
    if (!cloudsRef.current) return
    
    // Start with 0 opacity and fade in as user scrolls
    gsap.fromTo(cloudsRef.current, 
      { opacity: 0 },
      {
        opacity: 1,
        scrollTrigger: {
          trigger: document.body,
          start: '50% top',
          end: 'bottom top',
          scrub: 1,
        }
      }
    )
  }, [])
  
  return (
    <div ref={cloudsRef} className="absolute inset-0 opacity-0">
      <svg className="absolute bottom-0 w-full h-1/2" viewBox="0 0 1920 400" preserveAspectRatio="none">
        <defs>
          <linearGradient id="cloudGradient" x1="0%" y1="0%" x2="0%" y2="100%">
            <stop offset="0%" stopColor="rgba(255,255,255,0.1)" />
            <stop offset="100%" stopColor="rgba(255,255,255,0.3)" />
          </linearGradient>
        </defs>
        <path 
          d="M0,300 Q480,200 960,250 T1920,200 L1920,400 L0,400 Z" 
          fill="url(#cloudGradient)"
          className="animate-float"
        />
        <path 
          d="M0,250 Q480,150 960,200 T1920,150 L1920,400 L0,400 Z" 
          fill="url(#cloudGradient)"
          className="animate-float-delayed"
        />
      </svg>
    </div>
  )
}

function LightRays() {
  const raysRef = useRef<HTMLDivElement>(null)
  
  useEffect(() => {
    if (!raysRef.current) return
    
    // Fade in light rays in the final section
    gsap.fromTo(raysRef.current,
      { opacity: 0 },
      {
        opacity: 1,
        scrollTrigger: {
          trigger: document.body,
          start: '80% top',
          end: 'bottom top',
          scrub: 1,
        }
      }
    )
  }, [])
  
  return (
    <div ref={raysRef} className="absolute inset-0 opacity-0 pointer-events-none">
      <div className="absolute top-0 right-0 w-full h-full">
        <div className="absolute top-0 right-0 w-96 h-96 bg-gradient-radial from-yellow-200/20 via-yellow-100/10 to-transparent blur-3xl" />
        <div className="absolute top-20 right-20 w-64 h-64 bg-gradient-radial from-orange-200/20 via-orange-100/10 to-transparent blur-2xl" />
      </div>
    </div>
  )
}
