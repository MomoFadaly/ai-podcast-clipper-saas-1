/**
 * Adobe Layout System Constants
 * 
 * Constants and utilities for the Adobe-style layout system
 */
 
import type { DockZone, PanelGroup } from '~/types/adobe-layout';
 
// Panel size constraints
export const PANEL_CONSTRAINTS = {
  MIN_PANEL_SIZE: 50, // Minimum panel size in pixels - reduced from 150 to allow narrower panels
  MIN_ZONE_SIZE_PERCENT: 5, // Minimum zone size as percentage - reduced from 10
  MAX_ZONE_SIZE_PERCENT: 95, // Maximum zone size as percentage - increased from 90
  DEFAULT_PANEL_SIZE: 300, // Default panel size in pixels
  RESIZER_SIZE: 4, // Size of resize handles in pixels
} as const;
 
// Layout behavior constants
export const LAYOUT_BEHAVIOR = {
  SNAP_THRESHOLD: 20, // Pixels to snap to edges/other panels
  DRAG_THRESHOLD: 5, // Minimum drag distance to start drag operation
  ANIMATION_DURATION: 200, // Animation duration in milliseconds
  RESIZE_DEBOUNCE: 16, // Debounce resize events (60fps)
} as const;
 
/**
 * Apply size constraints to panel dimensions
 */
export function applyPanelSizeConstraints(
  size: number, 
  containerSize: number,
  minSize: number = PANEL_CONSTRAINTS.MIN_PANEL_SIZE
): number {
  const minPercent = (minSize / containerSize) * 100;
  const maxPercent = PANEL_CONSTRAINTS.MAX_ZONE_SIZE_PERCENT;
  
  return Math.max(minPercent, Math.min(maxPercent, size));
}
 
/**
 * Calculate minimum size as percentage based on container dimensions
 */
export function calculateMinSizePercent(containerSize: number): number {
  return (PANEL_CONSTRAINTS.MIN_PANEL_SIZE / containerSize) * 100;
}
 
/**
 * Normalize zone sizes to ensure they sum to 100%
 */
export function normalizeSizes(zones: Array<{ size?: number }>): void {
  if (zones.length === 0) return;
  
  const totalSize = zones.reduce((sum, zone) => sum + (zone.size || 0), 0);
  
  if (totalSize === 0 || Math.abs(totalSize - 100) > 0.1) {
    const equalSize = 100 / zones.length;
    zones.forEach(zone => {
      zone.size = equalSize;
    });
  }
}
 
/**
 * Calculate optimal panel group sizes based on content
 */
export function calculateOptimalSizes(
  children: Array<DockZone | PanelGroup>,
  containerSize: number
): Array<number> {
  const minSizePercent = calculateMinSizePercent(containerSize);
  const availableSpace = 100 - (minSizePercent * children.length);
  
  // If not enough space, use minimum sizes
  if (availableSpace <= 0) {
    return new Array(children.length).fill(minSizePercent);
  }
  
  // Distribute remaining space based on priority or equally
  const baseSize = minSizePercent;
  const extraSpace = availableSpace / children.length;
  
  return children.map(() => baseSize + extraSpace);
}
 
/**
 * Validate zone configuration for consistency
 */
export function validateZoneConfig(zone: DockZone): boolean {
  try {
    // Check for required properties
    if (!zone.id || !zone.orientation) {
      return false;
    }
    
    // Validate children
    if (zone.children) {
      for (const child of zone.children) {
        if ('type' in child) {
          // Panel group validation
          if (!child.id || child.type !== 'panel-group') {
            return false;
          }
        } else {
          // Recursive zone validation
          if (!validateZoneConfig(child)) {
            return false;
          }
        }
      }
    }
    
    return true;
  } catch (error) {
    console.error('Zone config validation error:', error);
    return false;
  }
}
 
/**
 * Deep clone a zone configuration
 */
export function cloneZoneConfig(zone: DockZone): DockZone {
  return JSON.parse(JSON.stringify(zone));
}
