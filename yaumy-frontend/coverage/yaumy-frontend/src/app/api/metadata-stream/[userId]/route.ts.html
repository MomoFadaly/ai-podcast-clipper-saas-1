import { type NextRequest } from 'next/server';
import { auth } from '~/server/auth';
import { db } from '~/server/db';

export async function GET(
  request: NextRequest,
  context: { params: Promise<{ userId: string }> }
) {
  try {
    const { userId } = await context.params;
    const session = await auth();

    console.log(`[SSE] Metadata stream request for user ${userId}, session user: ${session?.user?.id}`);

    if (!session || session.user?.id !== userId) {
      console.log(`[SSE] Unauthorized: session=${!!session}, userId match=${session?.user?.id === userId}`);
      return new Response('Unauthorized', { status: 401 });
    }

  const encoder = new TextEncoder();

  const stream = new ReadableStream({
    async start(controller) {
      // Send initial ping
      controller.enqueue(encoder.encode('data: {"type":"ping"}\n\n'));
      console.log(`[SSE] Started metadata stream for user ${userId}`);

      // Track sent updates to avoid duplicates
      const sentUpdates = new Set<string>();

      // Poll for metadata updates every 2 seconds
      const interval = setInterval(() => {
        void (async () => {
        try {
          // Find items that were recently updated
          const recentlyUpdated = await db.uploadedFile.findMany({
            where: {
              userId: userId,
              updatedAt: {
                gte: new Date(Date.now() - 15000), // Last 15 seconds
              },
              // Only get items that have actual metadata (not just the URL)
              AND: [
                {
                  OR: [
                    {
                      // YouTube videos
                      s3Key: {
                        contains: 'youtube',
                      },
                    },
                    {
                      // Or any URL-based item
                      s3Key: {
                        startsWith: 'http',
                      },
                    },
                  ],
                },
                {
                  // Must have at least one of these metadata fields populated
                  OR: [
                    {
                      // Has a real title (not just the URL)
                      displayName: {
                        not: {
                          equals: db.uploadedFile.fields.s3Key,
                        },
                      },
                    },
                    {
                      // Has duration
                      duration: {
                        gt: 0,
                      },
                    },
                    {
                      // Has a description
                      episodeDescription: {
                        not: null,
                      },
                    },
                  ],
                },
              ],
            },
            select: {
              id: true,
              displayName: true,
              thumbnailUrl: true,
              duration: true,
              episodeDescription: true,
              s3Key: true,
              status: true,
              updatedAt: true,
            },
          });

          if (recentlyUpdated.length > 0) {
            console.log(`[SSE] Found ${recentlyUpdated.length} updated items`);
          }
          
          for (const item of recentlyUpdated) {
            // Create a unique key for this update
            const updateKey = `${item.id}-${item.updatedAt.getTime()}`;
            
            // Skip if we've already sent this update
            if (sentUpdates.has(updateKey)) {
              continue;
            }
            
            const update = {
              itemId: item.id,
              metadata: {
                displayName: item.displayName,
                thumbnailUrl: item.thumbnailUrl,
                duration: item.duration,
                description: item.episodeDescription,
              },
            };

            console.log(`[SSE] Sending update for item ${item.id}:`, update.metadata.displayName);
            controller.enqueue(
              encoder.encode(`data: ${JSON.stringify(update)}\n\n`)
            );
            
            // Mark this update as sent
            sentUpdates.add(updateKey);
          }
        } catch (error) {
          console.error('Error polling for updates:', error);
        }
        })();
      }, 2000);

      // Clean up on disconnect
      request.signal.addEventListener('abort', () => {
        clearInterval(interval);
        controller.close();
      });
    },
  });

  return new Response(stream, {
    headers: {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache',
      'Connection': 'keep-alive',
    },
  });
  } catch (error) {
    console.error('[SSE] Error in metadata stream:', error);
    return new Response('Internal Server Error', { status: 500 });
  }
}
